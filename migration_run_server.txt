create table if not exists run_sessions (
  username text primary key references students(username) on delete cascade,
  status text not null default 'active',
  round int not null default 1,
  score int not null default 0,
  range_min int not null,
  range_max int not null,
  allowed int not null,
  secret int not null,
  moves int not null default 0,
  frost boolean not null default false,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);
alter table run_sessions enable row level security;

create or replace function run_cfg()
returns jsonb
language sql
stable
security definer
set search_path = public, pg_temp
as $$
  select coalesce(
    (select run_config from game_config limit 1),
    '{"start_range":10,"growth":2.2,"range_cap":2000,"buffer_start":5,
      "buffer_shrink":1,"squeeze_start":8,"min_attempts":3}'::jsonb);
$$;

create or replace function min_attempts(p_range int)
returns int
language sql
immutable
as $$
  select greatest(1, ceil(log(2::numeric, greatest(1, p_range)::numeric))::int);
$$;

create or replace function round_range_value(v double precision)
returns int
language sql
immutable
as $$
  select greatest(10, (round((v / step)::numeric) * step)::int)
  from (select case when v < 100 then 10 when v < 500 then 50
                    when v < 1000 then 100 when v < 5000 then 500 else 1000 end as step) s;
$$;

create or replace function run_range_for_round(n int)
returns int
language sql
stable
security definer
set search_path = public, pg_temp
as $$
  select round_range_value(least(
    (run_cfg() ->> 'start_range')::double precision
      * power((run_cfg() ->> 'growth')::double precision, (n - 1)::double precision),
    (run_cfg() ->> 'range_cap')::double precision));
$$;

create or replace function run_attempts_for_round(n int, p_range int)
returns int
language sql
stable
security definer
set search_path = public, pg_temp
as $$
  select greatest(
    (run_cfg() ->> 'min_attempts')::int,
    least(min_attempts(p_range) + greatest(0, (run_cfg() ->> 'buffer_start')::int
                                              - (n - 1) * (run_cfg() ->> 'buffer_shrink')::int),
          greatest(2, ceil(p_range * 0.6)::int))
    - greatest(0, n - (run_cfg() ->> 'squeeze_start')::int));
$$;

create or replace function run_deal_round(p_username text, p_round int, p_frost boolean)
returns void
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare size int; lo int; hi int; half int;
begin
  size := run_range_for_round(p_round);
  if p_frost then
    half := size / 2; lo := -half; hi := half;
  else
    lo := 1; hi := size;
  end if;
  update run_sessions
  set round = p_round,
      range_min = lo,
      range_max = hi,
      allowed = run_attempts_for_round(p_round, hi - lo + 1),
      secret = lo + floor(random() * (hi - lo + 1))::int,
      moves = 0,
      updated_at = now()
  where username = p_username;
end;
$$;

create or replace function run_finish(p_username text)
returns void
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare r run_sessions; cur_best int;
begin
  select * into r from run_sessions where username = p_username;
  if r.username is null or r.status <> 'active' then return; end if;

  insert into runs(username, score, rounds) values (r.username, r.score, greatest(0, r.round - 1));
  select best_run_score into cur_best from students where username = r.username;
  if r.score > coalesce(cur_best, 0) then
    update students set best_run_score = r.score, best_run_rounds = greatest(0, r.round - 1)
    where username = r.username;
  end if;
  update run_sessions set status = 'finished', updated_at = now() where username = p_username;
end;
$$;

create or replace function run_snapshot(p_username text)
returns jsonb
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare r run_sessions; best int;
begin
  select * into r from run_sessions where username = p_username;
  if r.username is null then return jsonb_build_object('active', false); end if;
  select best_run_score into best from students where username = p_username;
  return jsonb_build_object(
    'active', r.status = 'active',
    'round', r.round,
    'score', r.score,
    'rangeMin', r.range_min,
    'rangeMax', r.range_max,
    'allowed', r.allowed,
    'moves', r.moves,
    'frost', r.frost,
    'best', coalesce(best, 0));
end;
$$;

create or replace function run_start(p_username text, p_pin text, p_frost boolean, p_fresh boolean)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; r run_sessions;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  select * into r from run_sessions where username = me;
  if r.username is not null and r.status = 'active' and not coalesce(p_fresh, false) then
    return run_snapshot(me);
  end if;
  if r.username is not null and r.status = 'active' then
    perform run_finish(me);
  end if;

  insert into run_sessions(username, status, round, score, range_min, range_max,
                           allowed, secret, moves, frost)
  values (me, 'active', 1, 0, 1, 10, 5, 1, 0, coalesce(p_frost, false))
  on conflict (username) do update
    set status = 'active', round = 1, score = 0, moves = 0,
        frost = coalesce(p_frost, false), updated_at = now();
  perform run_deal_round(me, 1, coalesce(p_frost, false));
  return run_snapshot(me);
end;
$$;

create or replace function run_state(p_username text, p_pin text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;
  return run_snapshot(me);
end;
$$;

create or replace function run_guess(p_username text, p_pin text, p_guess int)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; r run_sessions; dist int; band int; span int; points int; out_secret int;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  select * into r from run_sessions where username = me for update;
  if r.username is null or r.status <> 'active' then raise exception 'no_run'; end if;
  if p_guess is null or p_guess < r.range_min or p_guess > r.range_max then
    raise exception 'out_of_range';
  end if;

  span := r.range_max - r.range_min + 1;
  dist := abs(p_guess - r.secret);
  band := feedback_tier(span, dist);
  update run_sessions set moves = r.moves + 1, updated_at = now() where username = me;
  r.moves := r.moves + 1;

  if dist = 0 then
    points := round(100.0 * r.round * (min_attempts(span)::numeric / r.moves));
    update run_sessions set score = r.score + points where username = me;
    perform run_deal_round(me, r.round + 1, r.frost);
    return jsonb_build_object('tier', band, 'roundWon', true, 'points', points,
                              'secret', r.secret, 'state', run_snapshot(me));
  end if;

  if r.moves >= r.allowed then
    out_secret := r.secret;
    perform run_finish(me);
    return jsonb_build_object('tier', band, 'over', true, 'secret', out_secret,
                              'score', r.score, 'rounds', greatest(0, r.round - 1),
                              'state', run_snapshot(me));
  end if;

  return jsonb_build_object('tier', band, 'state', run_snapshot(me));
end;
$$;

create or replace function run_give_up(p_username text, p_pin text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; r run_sessions;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;
  select * into r from run_sessions where username = me;
  if r.username is null or r.status <> 'active' then raise exception 'no_run'; end if;
  perform run_finish(me);
  return jsonb_build_object('score', r.score, 'rounds', greatest(0, r.round - 1),
                            'secret', r.secret);
end;
$$;

revoke execute on function run_cfg() from public, anon;
revoke execute on function run_deal_round(text, int, boolean) from public, anon;
revoke execute on function run_finish(text) from public, anon;
revoke execute on function run_snapshot(text) from public, anon;
grant execute on function min_attempts(int) to anon;
grant execute on function run_range_for_round(int) to anon;
grant execute on function run_attempts_for_round(int, int) to anon;
grant execute on function run_start(text, text, boolean, boolean) to anon;
grant execute on function run_state(text, text) to anon;
grant execute on function run_guess(text, text, int) to anon;
grant execute on function run_give_up(text, text) to anon;
