create or replace function tier_step(v double precision)
returns int
language sql
immutable
as $$
  select case
    when v < 12 then 1
    when v < 30 then 5
    when v < 100 then 10
    when v < 300 then 25
    when v < 1000 then 50
    when v < 3000 then 100
    else 500
  end;
$$;

create or replace function round_tier(v double precision, up boolean)
returns int
language sql
immutable
as $$
  select greatest(1, (
    case when up then ceil(v / tier_step(v))::numeric
         else round((v / tier_step(v))::numeric)
    end * tier_step(v))::int);
$$;

create or replace function tier_upper(p_span int)
returns int[]
language plpgsql
immutable
as $$
declare
  hot int[] := array[50, 20, 10, 3];
  md double precision;
  ratio double precision;
  u int[] := '{}';
  i int;
begin
  if p_span = 10 then return array[9, 7, 6, 5, 4, 3, 2, 1]; end if;
  if p_span = 100 then return array[100, 60, 30, 15, 8, 4, 2, 1]; end if;
  if p_span = 1000 then return array[1000, 500, 200, 100, 50, 20, 10, 3]; end if;

  md := greatest(1, p_span - 1);

  if md <= hot[1] then
    for i in 0..7 loop
      u := u || greatest(round_tier(md * power(0.45::double precision, i::double precision), i = 0), 8 - i);
    end loop;
  else
    ratio := power(hot[1]::double precision / md, 0.25::double precision);
    for i in 0..3 loop
      u := u || round_tier(md * power(ratio, i::double precision), i = 0);
    end loop;
    for i in 1..4 loop
      u := u || hot[i];
    end loop;
  end if;

  for i in 2..8 loop
    if u[i] >= u[i - 1] then u[i] := greatest(1, u[i - 1] - 1); end if;
  end loop;
  return u;
end;
$$;

create or replace function feedback_tier(p_span int, p_distance int)
returns int
language plpgsql
immutable
as $$
declare u int[]; i int; lo int; hi int;
begin
  if p_distance = 0 then return 8; end if;
  u := tier_upper(p_span);
  for i in 1..8 loop
    hi := u[i];
    lo := case when i = 8 then 1 else u[i + 1] + 1 end;
    if p_distance >= lo and p_distance <= hi then return i - 1; end if;
  end loop;
  return 0;
end;
$$;

revoke execute on function tier_step(double precision) from public, anon;
revoke execute on function round_tier(double precision, boolean) from public, anon;
revoke execute on function tier_upper(int) from public, anon;
revoke execute on function feedback_tier(int, int) from public, anon;
