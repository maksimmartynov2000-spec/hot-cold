-- Шесть питомцев — герои Приключения. Применять в Supabase: SQL Editor →
-- вставить целиком → Run. Применить ДО мёржа. Повторный запуск ничего не
-- ломает и ничего не возвращает второй раз.
--
-- • Питомцы теперь — герои миров: Барсук Очкарик и Филин Монокль (Волшебный
--   лес), Тукан Панама и Хамелеон Очки (Джунгли), Краб Матрос и Пингвин
--   Капитан (Морская глубина). Все шесть покупаются за монеты.
-- • Огнелис, Ледяной волк и Грозовой дракон и вещи для них (колпак мага, шлем
--   викинга, тёмные очки, шарф, бабочка, плащ) из игры уходят: купившим —
--   полная цена назад, в монетах или кристаллах, как платили. Корона и ауры
--   стихий остаются: они подходят новым питомцам.
-- • Облики (новый вид вещи skin) — за кристаллы, по пять у каждого питомца;
--   звёздный облик не продаётся, он открывается на 10-м уровне питомца.
-- • Вещи — за монеты: корона всем, по одной-две своих у каждого. Ауры дороже
--   вещей: четыре общие (пламя, иней, гроза, звёзды) и по две своих у каждого.
-- • Наряд хранится на каждого питомца отдельно: сменили питомца — у прежнего
--   всё осталось надетым. Места: голова, глаза, шея, тело, аура и облик.
-- • Вещь или облик одного питомца продаются только тому, у кого он есть; в
--   товары дня попадают только вещи своих питомцев.
-- Всё считает база: цену, скидку и опыт из игры не прислать.

-- Чья вещь: облик и вещь одного питомца — его код; корона и ауры стихий — всем (null)
create or replace function pet_item_for(p_kind text, p_code text)
returns text language sql immutable as $$
  select case
    when p_kind = 'skin' then split_part(p_code, '_', 1)
    when p_kind = 'gear' then case
      when p_code in ('mortar', 'starglasses', 'aura_leaves', 'aura_fireflies') then 'badger'
      when p_code in ('tophat', 'argyle', 'aura_moon', 'aura_feathers') then 'owl'
      when p_code in ('lei', 'sunglasses', 'aura_tropics', 'aura_fruit') then 'toucan'
      when p_code in ('cap', 'safari', 'aura_rainbow', 'aura_butterflies') then 'cham'
      when p_code in ('lifebuoy', 'striped', 'aura_bubbles', 'aura_seabed') then 'crab'
      when p_code in ('jacket', 'binoculars', 'aura_snow', 'aura_fish') then 'penguin'
    end
  end;
$$;

-- Куда надевается вещь питомца: голова, глаза, шея, тело, аура
create or replace function pet_gear_slot(p_code text)
returns text language sql immutable as $$
  select case
    when p_code in ('crown', 'mortar', 'tophat', 'cap', 'safari') then 'head'
    when p_code in ('starglasses', 'sunglasses') then 'eyes'
    when p_code in ('argyle', 'lei', 'binoculars') then 'neck'
    when p_code in ('lifebuoy', 'striped', 'jacket') then 'body'
    when p_code like 'aura\_%' then 'aura'
  end;
$$;

-- Звёздный облик открывается на 10-м уровне: 2700 опыта (как petLevel в js/pets.js)
create or replace function pet_star_xp()
returns int language sql immutable as $$ select 2700 $$;

-- ============ 4. Товары дня ============
-- Три вещи со скидкой 40, 25 и 15 % на сегодня: из того, чего у игрока нет,
-- кроме питомцев и вещей чужих питомцев. Выбираются один раз за день
create or replace function shop_deals_today(p_username text)
returns jsonb
language plpgsql security definer set search_path = public, pg_temp as $$
declare d date; got jsonb;
begin
  d := player_day(p_username);
  select items into got from shop_deals where username = p_username and day = d;
  if got is not null then return got; end if;
  select coalesce(jsonb_agg(jsonb_build_object('kind', x.kind, 'code', x.code, 'pct', (array[40, 25, 15])[x.n]) order by x.n), '[]'::jsonb)
    into got
    from (select c.kind, c.code, (row_number() over (order by random()))::int as n
            from shop_catalog() c
           where c.kind <> 'pet' and not cosmetic_has(p_username, c.kind, c.code)
             and (pet_item_for(c.kind, c.code) is null or cosmetic_has(p_username, 'pet', pet_item_for(c.kind, c.code)))) x
   where x.n <= 3;
  insert into shop_deals(username, day, items) values (p_username, d, got) on conflict do nothing;
  delete from shop_deals where username = p_username and day < d - 7;
  select items into got from shop_deals where username = p_username and day = d;
  return got;
end;
$$;

-- ============ 5. Покупка ============
-- Как в 063, и ещё: вещь или облик одного питомца — только тому, у кого он есть
create or replace function shop_buy(p_username text, p_pin text, p_kind text, p_code text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; pr record; w wallet; who text;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;
  if not exists (select 1 from shop_catalog() c where c.kind = p_kind and c.code = p_code) then raise exception 'no_such_item'; end if;
  insert into wallet(username) values (me) on conflict do nothing;
  select * into w from wallet where username = me for update;
  if cosmetic_has(me, p_kind, p_code) then raise exception 'already_owned'; end if;
  who := pet_item_for(p_kind, p_code);
  if who is not null and not cosmetic_has(me, 'pet', who) then raise exception 'pet_locked'; end if;
  select * into pr from shop_price(me, p_kind, p_code);
  if w.coins < pr.coins or w.gems < pr.gems then raise exception 'not_enough'; end if;
  update wallet set coins = coins - pr.coins, gems = gems - pr.gems, updated_at = now()
   where username = me returning * into w;
  insert into wallet_log(username, coins, gems, reason, ref, day)
  values (me, -pr.coins, -pr.gems, 'buy', p_kind || ':' || p_code, player_day(me));
  perform cosmetic_grant(me, p_kind, p_code, 'shop');
  if p_kind = 'pet' then update students set pet = p_code where username = me and pet is null; end if;
  if p_kind = 'thermo' then update students set thermo = p_code where username = me and thermo is null; end if;
  if p_kind = 'winfx' then update students set winfx = p_code where username = me and winfx is null; end if;
  return jsonb_build_object('coins', w.coins, 'gems', w.gems, 'kind', p_kind, 'code', p_code, 'paid', pr.coins + pr.gems, 'pct', pr.pct);
end;
$$;

-- ============ 6. Наряд: на каждого питомца свой ============
-- pet_gear у игрока: {"badger": {"head": "crown", "skin": "honey"}, "owl": {...}}.
-- Надеть на выбранного питомца вещь (голова, глаза, шея, тело, аура) или
-- облик (skin: свой купленный или star с 10-го уровня), снять — null.
-- Возвращает наряды всех питомцев
create or replace function set_pet_gear(p_username text, p_pin text, p_slot text, p_code text)
returns jsonb language plpgsql security definer set search_path = public, extensions, pg_temp as $$
declare me text; p text; g jsonb; o jsonb;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;
  select pet, coalesce(pet_gear, '{}'::jsonb) into p, g from students where username = me;
  if p is null or not cosmetic_has(me, 'pet', p) then raise exception 'no_pet'; end if;
  if p_slot is null or p_slot not in ('head', 'eyes', 'neck', 'body', 'aura', 'skin') then raise exception 'bad_slot'; end if;
  if p_code is not null then
    if p_slot = 'skin' then
      if p_code = 'star' then
        if coalesce((select x.xp from pet_xp x where x.username = me and x.pet = p), 0) < pet_star_xp() then raise exception 'skin_locked'; end if;
      elsif not cosmetic_has(me, 'skin', p || '_' || p_code) then raise exception 'skin_locked'; end if;
    else
      if pet_gear_slot(p_code) is distinct from p_slot then raise exception 'bad_slot'; end if;
      if not cosmetic_has(me, 'gear', p_code) then raise exception 'gear_locked'; end if;
      if pet_item_for('gear', p_code) is not null and pet_item_for('gear', p_code) <> p then raise exception 'gear_locked'; end if;
    end if;
  end if;
  o := coalesce(g -> p, '{}'::jsonb);
  o := case when p_code is null then o - p_slot else o || jsonb_build_object(p_slot, p_code) end;
  g := g || jsonb_build_object(p, o);
  update students set pet_gear = g where username = me;
  return g;
end;
$$;

-- ============ 7. Прежние питомцы и вещи: монеты назад ============
-- Купившим Огнелиса, волка, дракона и вещи для них — полная цена назад.
-- Второй раз не вернётся: запись о возврате с той же ссылкой в wallet_log
-- вставляется только однажды, а сами вещи из коллекции удаляются.
-- Прежние наряды снимаются: места изменились, а вещи теперь у каждого свои.
-- Возвращает, за сколько вещей вернулись деньги
create or replace function six_pets_cleanup()
returns int language plpgsql security definer set search_path = public, pg_temp as $$
declare r record; n int; total int := 0;
begin
  for r in
    select c.username, c.kind, c.code, o.coins, o.gems
      from cosmetics c
      join (values ('pet', 'fox', 450, 0), ('pet', 'wolf', 450, 0), ('pet', 'drake', 0, 30),
                   ('gear', 'wizard', 360, 0), ('gear', 'viking', 220, 0), ('gear', 'shades', 180, 0),
                   ('gear', 'scarf', 100, 0), ('gear', 'bowtie', 80, 0), ('gear', 'cape', 400, 0)) as o(kind, code, coins, gems)
        on o.kind = c.kind and o.code = c.code
  loop
    insert into wallet(username) values (r.username) on conflict do nothing;
    insert into wallet_log(username, coins, gems, reason, ref, day)
    values (r.username, r.coins, r.gems, 'refund', r.kind || ':' || r.code, player_day(r.username))
    on conflict do nothing;
    get diagnostics n = row_count;
    if n > 0 then
      update wallet set coins = coins + r.coins, gems = gems + r.gems, updated_at = now() where username = r.username;
      total := total + 1;
    end if;
    delete from cosmetics where username = r.username and kind = r.kind and code = r.code;
  end loop;
  update students set pet = null where pet in ('fox', 'wolf', 'drake');
  -- Прежний наряд лежал по местам ({"head": …}), новый — по питомцам ({"badger": {…}})
  update students set pet_gear = '{}'::jsonb
   where pet_gear <> '{}'::jsonb and not (pet_gear ? 'badger' or pet_gear ? 'owl' or pet_gear ? 'toucan' or pet_gear ? 'cham' or pet_gear ? 'crab' or pet_gear ? 'penguin');
  delete from pet_xp where pet in ('fox', 'wolf', 'drake');
  -- Товары дня с ушедшими вещами выбираются заново
  delete from shop_deals s
   where exists (select 1 from jsonb_array_elements(s.items) x
                  where not exists (select 1 from shop_catalog() c where c.kind = x->>'kind' and c.code = x->>'code'));
  return total;
end;
$$;
select six_pets_cleanup();

-- ============ 8. Права ============
grant execute on function shop_buy(text, text, text, text) to anon;
grant execute on function set_pet_gear(text, text, text, text) to anon;
revoke execute on function shop_deals_today(text) from public, anon;
revoke execute on function six_pets_cleanup() from public, anon;
