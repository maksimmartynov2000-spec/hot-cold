-- Туман, слепота и короткая память — на весь раунд. Применять в Supabase:
-- SQL Editor → вставить целиком → Run. Применить ДО мёржа.
--
-- Раньше помехи прятали ходы только пока ходит сам игрок. Пока ходил
-- соперник, игрок под туманом видел его ходы и подсказки, а на своём ходу они
-- превращались в «•••» — но он их уже видел, и туман почти ничего не делал.
-- Теперь помеха действует весь раунд, на любом ходу.
--
-- Прячутся только ходы, сделанные после того, как помеха легла: ход, которым
-- соперник взял туман, и всё до него игрок уже видел — прятать их поздно.
-- Бот под помехой видит ровно то же, что видел бы человек.

-- 1. С какого хода действует помеха: номер последнего хода на тот момент.
--    Ходы с номером больше — закрыты. 0 — закрыто всё (так ведут себя партии,
--    где помеха легла до этой миграции)
alter table matches add column if not exists fog_from bigint[] not null default array[0, 0]::bigint[];
alter table matches add column if not exists blind_from bigint[] not null default array[0, 0]::bigint[];

-- 2. Отметку ставит сама база, в момент, когда помеха ложится, — какая бы
--    функция её ни наложила. Ход, которым помеху взяли, к этому моменту уже
--    записан, поэтому он остаётся открытым
create or replace function matches_hide_from()
returns trigger
language plpgsql
set search_path = public, pg_temp
as $$
declare last_id bigint; i int;
begin
  if new.fog is distinct from old.fog or new.blind is distinct from old.blind then
    select coalesce(max(x.id), 0) into last_id from match_moves x where x.match_id = new.id;
    for i in 1..2 loop
      if coalesce(new.fog[i], false) and not coalesce(old.fog[i], false) then
        new.fog_from[i] := last_id;
      end if;
      if coalesce(new.blind[i], false) and not coalesce(old.blind[i], false) then
        new.blind_from[i] := last_id;
      end if;
    end loop;
  end if;
  return new;
end;
$$;

drop trigger if exists matches_hide_from on matches;
create trigger matches_hide_from before update of fog, blind on matches
for each row execute function matches_hide_from();

-- 3. Что видит игрок: помехи — весь раунд, закрыты ходы после отметки
create or replace function match_state(p_username text, p_pin text, p_match_id bigint)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; m matches; my_seat int; moves jsonb; reveal boolean;
        visible int; hide_mine boolean; hide_theirs boolean; chat jsonb; my_elo int;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  perform expire_stale_matches();

  select * into m from matches where id = p_match_id;
  if m.id is null then raise exception 'no_such_match'; end if;
  my_seat := match_seat(m, me);
  if my_seat is null then raise exception 'not_your_match'; end if;

  -- Партии людей бот не трогает: лишняя блокировка строки на каждом опросе
  if m.bot_seat is not null then perform bot_act(p_match_id); end if;
  perform apply_turn_timeout(p_match_id);

  select * into m from matches where id = p_match_id;

  reveal := m.round_over or m.match_over or m.status <> 'active';
  -- Помехи действуют весь раунд, чей бы ни был ход. Прячутся только ходы,
  -- сделанные после того, как помеха легла: прежние игрок уже видел
  hide_mine := (not reveal) and m.blind[my_seat + 1];
  hide_theirs := (not reveal) and m.fog[my_seat + 1];
  visible := case when (not reveal) and m.short_memory[my_seat + 1] then 2 else 4 end;

  select coalesce(jsonb_agg(x order by x_id), '[]'::jsonb) into moves from (
    select mv.id as x_id,
           case
             when mv.kind = 'timeout'
               then jsonb_build_object('seat', mv.seat, 'timeout', true)
             when (mv.seat = my_seat and hide_mine and mv.id > m.blind_from[my_seat + 1])
               or (mv.seat <> my_seat and hide_theirs and mv.id > m.fog_from[my_seat + 1])
               then jsonb_build_object('seat', mv.seat, 'hidden', true)
             else jsonb_build_object('seat', mv.seat, 'guess', mv.guess, 'tier', mv.tier)
           end as x
    from (select * from match_moves mv2
          where mv2.match_id = p_match_id and mv2.round = m.round
          order by mv2.id desc limit visible) mv
  ) q;

  -- Фразы отдаём вместе с возрастом: клиент сам решит, показывать ли ещё
  select coalesce(jsonb_agg(jsonb_build_object(
           'id', c.id, 'seat', c.seat, 'code', c.code,
           'ago', floor(extract(epoch from (now() - c.created_at)))::int) order by c.id), '[]'::jsonb)
    into chat
  from (select * from match_chat c2 where c2.match_id = p_match_id
        order by c2.id desc limit 6) c;

  select e.elo into my_elo from elo_ratings e
  where e.username = me and e.mode = coalesce(m.ranked_mode, 0);

  return jsonb_build_object(
    'id', m.id, 'seat', my_seat,
    'names', jsonb_build_array(m.p0, m.p1),
    'status', m.status, 'winsNeeded', m.wins_needed,
    'rangeMin', m.range_min, 'rangeMax', m.range_max, 'frost', m.frost,
    'round', m.round, 'wins', m.wins, 'cur', m.cur, 'tokens', m.tokens, 'armed', m.armed,
    'roundOver', m.round_over, 'roundWinner', m.round_winner, 'matchOver', m.match_over,
    'moves', moves,
    'secret', case when reveal then to_jsonb(m.secret) else 'null'::jsonb end,
    'bonusesOn', m.bonuses_on,
    'fog', m.fog, 'blind', m.blind, 'autoLava', m.auto_lava,
    'skip', m.skip_turn, 'shortMemory', m.short_memory, 'rush', m.rush,
    'nearBonus', m.near_bonus, 'lastBonus', m.last_bonus, 'lastBonusBy', m.last_bonus_by,
    'lastTimeout', m.last_timeout,
    'forced', forced_turn_kind(m),
    'turnSeconds', turn_seconds(m),
    'secondsLeft', case when m.turn_deadline is null then null
                        else greatest(0, ceil(extract(epoch from (m.turn_deadline - now()))))::int end,
    'chat', chat,
    'ranked', m.ranked,
    'rankedMode', m.ranked_mode,
    'botSeat', m.bot_seat,
    'forfeitBy', m.forfeit_by,
    'eloDelta', case when m.elo_applied then to_jsonb(m.elo_delta) else 'null'::jsonb end,
    'elo', coalesce(my_elo, 1000),
    'updatedAt', m.updated_at
  );
end;
$$;

-- 4. Что видит бот: то же правило
create or replace function bot_candidates(p_match_id bigint, p_seat int, p_careless boolean)
returns int[]
language plpgsql
stable
security definer
set search_path = public, pg_temp
as $$
declare m matches; db int[]; mv record; cand boolean[]; n int; i int; res int[] := '{}';
begin
  select * into m from matches where id = p_match_id;
  n := match_span(m);
  db := bot_dist_bands(n);
  cand := array_fill(true, array[n]);
  for mv in
    select k.guess, k.tier from (
      select x.guess, x.tier, x.id from match_moves x
      where x.match_id = p_match_id and x.round = m.round
        and x.guess is not null and x.tier is not null
        and not (m.fog[p_seat + 1] and x.seat <> p_seat and x.id > m.fog_from[p_seat + 1])
        and not (m.blind[p_seat + 1] and x.seat = p_seat and x.id > m.blind_from[p_seat + 1])
      order by x.id desc
      limit case when p_careless then 1 when m.short_memory[p_seat + 1] then 2 else null end
    ) k
  loop
    for i in 1..n loop
      if cand[i] and db[abs(m.range_min + i - 1 - mv.guess) + 1] <> mv.tier then
        cand[i] := false;
      end if;
    end loop;
  end loop;
  for i in 1..n loop
    if cand[i] then res := res || (m.range_min + i - 1); end if;
  end loop;
  -- Ответы честные, поэтому загаданное всегда среди возможных и пусто не
  -- бывает. На всякий случай — весь диапазон, а не ошибка посреди партии
  if coalesce(array_length(res, 1), 0) = 0 then
    select array_agg(g order by g) into res from generate_series(m.range_min, m.range_max) g;
  end if;
  return res;
end;
$$;

revoke execute on function matches_hide_from() from public, anon;
revoke execute on function bot_candidates(bigint, int, boolean) from public, anon;
grant execute on function match_state(text, text, bigint) to anon;
