-- Мгновенные ходы и сообщения через Supabase Realtime. Применять в Supabase:
-- SQL Editor → вставить целиком → Run. Можно и после мёржа: без неё игра
-- работает как раньше, на опросе.
--
-- Раньше ход соперника доходил до экрана за время до 2 секунд, заявка в
-- друзья и сообщение — до 10 и 5 секунд: игра спрашивала базу по таймеру.
-- Теперь база сама толкает сигнал «что-то изменилось», и игра тут же
-- спрашивает, что именно. В сигнале нет ничего, кроме вида и номера: ни ходов,
-- ни текста — всё по-прежнему читается через функции с PIN.
--
-- Канал у каждого игрока свой, с именем из случайного ключа: чужой канал не
-- угадать. Realtime недоступен — сигнала нет, игра живёт на опросе, как раньше.

-- 1. Ключ канала
alter table students add column if not exists signal_key text;
create unique index if not exists students_signal_key_idx on students (signal_key);

create or replace function my_signal_key(p_username text, p_pin text)
returns text
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; k text;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;
  select signal_key into k from students where username = me;
  if k is null then
    k := encode(extensions.gen_random_bytes(16), 'hex');
    update students set signal_key = k where username = me;
  end if;
  return k;
end;
$$;

-- 2. Отправка сигнала. Ошибка Realtime игру не ломает: ход уже записан,
--    а соперник увидит его на следующем опросе
create or replace function hc_signal(p_username text, p_kind text, p_id bigint)
returns void
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare k text;
begin
  if p_username is null or p_username like '@bot:%' or p_username like '#%' then return; end if;
  if to_regprocedure('realtime.send(jsonb,text,text,boolean)') is null then return; end if;
  select signal_key into k from students where username = p_username;
  if k is null then return; end if;
  begin
    execute 'select realtime.send($1, $2, $3, false)'
      using jsonb_build_object('k', p_kind, 'id', p_id), 'sig', 'hc:' || k;
  exception when others then
    null;
  end;
end;
$$;

-- 3. Откуда сигналы
-- Партия: ход, чат, смена хода, раунда, старта, вызов и ответ на него.
-- Отметка «я на месте» пишется при каждом опросе — на неё сигнала нет,
-- иначе два экрана будили бы друг друга без конца
create or replace function matches_signal()
returns trigger
language plpgsql
security definer
set search_path = public, pg_temp
as $$
begin
  perform hc_signal(new.p0, 'match', new.id);
  perform hc_signal(new.p1, 'match', new.id);
  return null;
end;
$$;
drop trigger if exists matches_signal_ins on matches;
create trigger matches_signal_ins after insert on matches
  for each row execute function matches_signal();
drop trigger if exists matches_signal_upd on matches;
create trigger matches_signal_upd after update on matches
  for each row when ((old.status, old.cur, old.round, old.round_over, old.match_over, old.lobby,
                      old.start_at, old.ready, old.armed, old.tokens, old.wins)
                     is distinct from
                     (new.status, new.cur, new.round, new.round_over, new.match_over, new.lobby,
                      new.start_at, new.ready, new.armed, new.tokens, new.wins))
  execute function matches_signal();

create or replace function match_rows_signal()
returns trigger
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare m matches;
begin
  select * into m from matches where id = new.match_id;
  if m.id is null then return null; end if;
  perform hc_signal(m.p0, 'match', m.id);
  perform hc_signal(m.p1, 'match', m.id);
  return null;
end;
$$;
drop trigger if exists match_moves_signal on match_moves;
create trigger match_moves_signal after insert on match_moves
  for each row execute function match_rows_signal();
drop trigger if exists match_chat_signal on match_chat;
create trigger match_chat_signal after insert on match_chat
  for each row execute function match_rows_signal();

-- Друзья: заявка, ответ, удаление
create or replace function friendships_signal()
returns trigger
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare r friendships;
begin
  r := case when tg_op = 'DELETE' then old else new end;
  perform hc_signal(r.requester, 'friends', null);
  perform hc_signal(r.addressee, 'friends', null);
  return null;
end;
$$;
drop trigger if exists friendships_signal on friendships;
create trigger friendships_signal after insert or update or delete on friendships
  for each row execute function friendships_signal();

-- Переписка: сигнал тому, кому написали
create or replace function friend_chat_signal()
returns trigger
language plpgsql
security definer
set search_path = public, pg_temp
as $$
begin
  perform hc_signal(case when new.sender = new.a then new.b else new.a end, 'chat', new.id);
  return null;
end;
$$;
drop trigger if exists friend_chat_signal on friend_chat;
create trigger friend_chat_signal after insert on friend_chat
  for each row execute function friend_chat_signal();

revoke execute on function hc_signal(text, text, bigint) from public, anon;
revoke execute on function matches_signal() from public, anon;
revoke execute on function match_rows_signal() from public, anon;
revoke execute on function friendships_signal() from public, anon;
revoke execute on function friend_chat_signal() from public, anon;
grant execute on function my_signal_key(text, text) to anon;
