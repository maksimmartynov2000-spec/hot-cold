-- Разбор онлайн-партии. Применять в Supabase: SQL Editor → вставить
-- целиком → Run. Применить ДО мёржа: новая игра спрашивает у базы все
-- ходы законченной партии.
--
-- Во время раунда игрок видит только последние ходы, и так и остаётся. Когда
-- партия закончена, скрывать уже нечего: все ходы всех раундов отдаются
-- обоим участникам, и игра разбирает их так же, как игру с другом.
--
-- Ход, который игрок не выбирал (бросок в лаву), и ход под помехой (туман,
-- слепота, короткая память) в разборе не оцениваются. Раньше база их не
-- различала — теперь помечает. У ходов, сделанных до этой миграции, пометок
-- нет: они оцениваются как обычные.

-- 1. Пометки на ходах
alter table match_moves add column if not exists forced boolean not null default false;
alter table match_moves add column if not exists limited boolean not null default false;

-- 2. Ход записывается с пометкой «под помехой»
create or replace function match_guess_core(p_match_id bigint, p_seat int, p_guess int)
returns int
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare m matches; seat int; dist int; band int; used_double boolean;
        hit record; btype text; radius int; done boolean;
begin
  seat := p_seat;
  select * into m from matches where id = p_match_id for update;
  if m.id is null then raise exception 'no_such_match'; end if;
  if m.status <> 'active' then raise exception 'not_playing'; end if;
  if m.round_over or m.match_over then raise exception 'round_over'; end if;
  if m.cur <> seat then raise exception 'not_your_turn'; end if;
  if m.skip_turn[seat + 1] or m.auto_lava[seat + 1] then raise exception 'forced_turn'; end if;
  if p_guess is null or p_guess < m.range_min or p_guess > m.range_max then
    raise exception 'out_of_range';
  end if;

  dist := abs(p_guess - m.secret);
  band := feedback_tier(match_span(m), dist);
  -- Для разбора: ход сделан под помехой — игрок видел не все ходы
  insert into match_moves(match_id, round, seat, guess, tier, limited)
  values (p_match_id, m.round, seat, p_guess, band,
          m.fog[seat + 1] or m.blind[seat + 1] or m.short_memory[seat + 1]);

  radius := bonus_near_radius(match_span(m));
  select * into hit from match_bonuses b
  where b.match_id = p_match_id and b.round = m.round and not b.taken and b.value = p_guess;

  btype := null;
  if hit.type is not null then
    btype := hit.type;
    if match_bonus_blocked(m, btype, 1 - seat) then btype := 'extra'; end if;
    update match_bonuses set taken = true
    where match_id = p_match_id and round = m.round and value = p_guess;
  end if;

  -- Сходил — значит на месте: счётчик молчания обнуляется
  update matches
  set last_bonus = btype,
      last_bonus_by = seat,
      last_timeout = false,
      timeouts[seat + 1] = 0,
      near_bonus = (btype is null and exists (
        select 1 from match_bonuses b
        where b.match_id = p_match_id and b.round = m.round and not b.taken
          and abs(b.value - p_guess) <= radius))
  where id = p_match_id;

  if btype is not null then perform apply_match_bonus(p_match_id, btype, seat); end if;

  select * into m from matches where id = p_match_id;
  used_double := m.armed;

  if dist = 0 then
    done := (m.wins[seat + 1] + 1) >= m.wins_needed;
    update matches
    set round_over = true, round_winner = seat,
        wins[seat + 1] = m.wins[seat + 1] + 1,
        match_over = done,
        status = case when done then 'finished' else status end,
        armed = false, turn_deadline = null, updated_at = now()
    where id = p_match_id;
    if done then perform apply_elo(p_match_id, seat); end if;
  elsif used_double then
    update matches set armed = false, updated_at = now() where id = p_match_id;
    select * into m from matches where id = p_match_id;
    update matches set turn_deadline = now() + make_interval(secs => turn_seconds(m))
    where id = p_match_id;
  else
    perform pass_turn(p_match_id, false);
  end if;

  return band;
end;
$$;

-- 3. Бросок в лаву помечается как ход без выбора
create or replace function forced_turn_core(p_match_id bigint, p_seat int)
returns text
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare m matches; seat int; kind text; v int;
begin
  seat := p_seat;
  select * into m from matches where id = p_match_id for update;
  if m.id is null then raise exception 'no_such_match'; end if;
  kind := forced_turn_kind(m);
  if kind is null then raise exception 'nothing_forced'; end if;
  if m.cur <> seat then raise exception 'not_your_turn'; end if;

  if kind = 'skip' then
    update matches set skip_turn[seat + 1] = false where id = p_match_id;
    if m.armed then
      -- Пропуск засчитан первым ходом: жетон тратится, ход остаётся за игроком
      update matches set armed = false, last_timeout = false, updated_at = now()
      where id = p_match_id;
      select * into m from matches where id = p_match_id;
      update matches set turn_deadline = now() + make_interval(secs => turn_seconds(m))
      where id = p_match_id;
      return 'skip';
    end if;
    perform pass_turn(p_match_id, false);
    return 'skip';
  end if;

  update matches set auto_lava[seat + 1] = false where id = p_match_id;
  select * into m from matches where id = p_match_id;
  v := lava_value(m);
  if v is null then
    perform pass_turn(p_match_id, false);
    return 'lava';
  end if;
  -- Бросок идёт обычным ходом, поэтому взведённый жетон там же и сработает
  perform match_guess_core(p_match_id, seat, v);
  -- Для разбора: этот ход игрок не выбирал
  update match_moves set forced = true
  where id = (select max(x.id) from match_moves x where x.match_id = p_match_id and x.seat = p_seat);
  return 'lava';
end;
$$;

-- 4. Все ходы законченной партии — только её участникам
create or replace function match_review(p_username text, p_pin text, p_match_id bigint)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; m matches; moves jsonb;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  select * into m from matches where id = p_match_id;
  if m.id is null then raise exception 'no_such_match'; end if;
  if match_seat(m, me) is null then raise exception 'not_your_match'; end if;
  -- Пока партия идёт, ходы прошлых раундов — подсказка о том, как играет
  -- соперник, а текущего — прямая подсказка. Только после конца
  if not m.match_over then raise exception 'match_not_over'; end if;

  select coalesce(jsonb_agg(jsonb_build_object(
           'round', mv.round, 'seat', mv.seat, 'guess', mv.guess, 'tier', mv.tier,
           'timeout', mv.kind = 'timeout', 'forced', mv.forced, 'limited', mv.limited)
         order by mv.id), '[]'::jsonb)
    into moves
  from match_moves mv where mv.match_id = p_match_id;

  return jsonb_build_object('id', m.id, 'moves', moves);
end;
$$;

revoke execute on function match_guess_core(bigint, int, int) from public, anon;
revoke execute on function forced_turn_core(bigint, int) from public, anon;
grant execute on function match_review(text, text, bigint) to anon;
