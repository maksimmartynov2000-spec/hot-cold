-- Уведомление «соперник принял вызов» и жетон до старта. Применять в
-- Supabase: SQL Editor → вставить целиком → Run.
--
-- Партия с человеком теперь ждёт обоих (миграция 035). Вызвавший мог не знать,
-- что вызов приняли, — и через 5 минут партия отменялась. Теперь ему приходит
-- уведомление. Заодно база знает ещё два вида уведомлений: напоминание о конце
-- сезона и «друг пришёл по вашему приглашению» — их включают следующие
-- миграции. Тексты уведомлений живут в функции-отправителе push/index.ts:
-- её нужно обновить в Supabase (шаги — в отчёте).

-- 1. Новые виды уведомлений
alter table push_outbox drop constraint if exists push_kind;
alter table push_outbox add constraint push_kind
  check (kind in ('friend_request', 'challenge', 'talk', 'accepted', 'season_end', 'invite_joined'));

-- 2. Вызов или реванш приняли — сказать вызвавшему
create or replace function push_on_match_accept()
returns trigger
language plpgsql
security definer
set search_path = public, pg_temp
as $$
begin
  if old.status = 'invited' and new.status = 'active' and new.bot_seat is null then
    insert into push_outbox(username, kind, who) values (new.p0, 'accepted', new.p1);
  end if;
  return new;
end;
$$;

drop trigger if exists push_match_accept on matches;
create trigger push_match_accept after update of status on matches
  for each row execute function push_on_match_accept();

-- 3. Жетон нельзя взвести, пока партия не началась
create or replace function use_match_token(p_username text, p_pin text,
                                           p_match_id bigint, p_arm boolean)
returns boolean
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; m matches; seat int;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  select * into m from matches where id = p_match_id;
  if m.id is null then raise exception 'no_such_match'; end if;
  seat := match_seat(m, me);
  if seat is null then raise exception 'not_your_match'; end if;
  if m.status <> 'active' or m.round_over then raise exception 'not_playing'; end if;
  if m.cur <> seat then raise exception 'not_your_turn'; end if;
  -- Пока ждём обоих и идёт отсчёт, жетон не взводится: партии ещё нет
  if m.lobby or (m.start_at is not null and m.start_at > now()) then
    raise exception 'not_started';
  end if;

  if not p_arm then
    if not m.armed then return false; end if;
    update matches set armed = false, tokens[seat + 1] = m.tokens[seat + 1] + 1,
                       updated_at = now()
    where id = p_match_id;
    return false;
  end if;

  if m.armed then return true; end if;
  if m.tokens[seat + 1] <= 0 then raise exception 'no_tokens'; end if;

  update matches set armed = true, tokens[seat + 1] = m.tokens[seat + 1] - 1,
                     updated_at = now()
  where id = p_match_id;
  return true;
end;
$$;

revoke execute on function push_on_match_accept() from public, anon;
grant execute on function use_match_token(text, text, bigint, boolean) to anon;
