create table if not exists friendships (
  requester text not null references students(username) on delete cascade,
  addressee text not null references students(username) on delete cascade,
  status text not null default 'pending',
  created_at timestamptz not null default now(),
  decided_at timestamptz,
  primary key (requester, addressee),
  constraint friendship_not_self check (requester <> addressee),
  constraint friendship_status check (status in ('pending', 'accepted', 'declined'))
);
create index if not exists friendships_addressee_idx on friendships (addressee, status);
alter table friendships enable row level security;

create or replace function friend_status(p_a text, p_b text)
returns text
language sql
stable
security definer
set search_path = public, pg_temp
as $$
  select case
    when f.status = 'accepted' then 'friend'
    when f.status = 'pending' and f.requester = p_a then 'outgoing'
    when f.status = 'pending' then 'incoming'
    when f.status = 'declined' and f.requester = p_a then 'declined'
    else null
  end
  from friendships f
  where (f.requester = p_a and f.addressee = p_b)
     or (f.requester = p_b and f.addressee = p_a)
  limit 1;
$$;

create or replace function find_students(p_username text, p_pin text, p_query text)
returns table(username text, best_run_score int, relation text)
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; q text;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  q := lower(trim(coalesce(p_query, '')));
  if length(q) < 2 then raise exception 'query_too_short'; end if;

  return query
  select s.username, s.best_run_score, friend_status(me, s.username)
  from students s
  where s.username_lower like q || '%'
    and s.username <> me
  order by s.username
  limit 10;
end;
$$;

create or replace function send_friend_request(p_username text, p_pin text, p_to text)
returns text
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; other text; cur text; pending_out int;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  select s.username into other from students s where s.username_lower = lower(trim(p_to));
  if other is null then raise exception 'no_such_student'; end if;
  if other = me then raise exception 'cannot_friend_self'; end if;

  cur := friend_status(me, other);
  if cur = 'friend' then return 'friend'; end if;
  if cur = 'outgoing' then return 'outgoing'; end if;

  if cur = 'incoming' then
    update friendships set status = 'accepted', decided_at = now()
    where requester = other and addressee = me;
    return 'friend';
  end if;

  if exists (select 1 from friendships f
             where f.requester = me and f.addressee = other
               and f.status = 'declined' and f.decided_at > now() - interval '24 hours') then
    raise exception 'recently_declined';
  end if;

  select count(*) into pending_out from friendships f
  where f.requester = me and f.status = 'pending';
  if pending_out >= 20 then raise exception 'too_many_requests'; end if;

  insert into friendships(requester, addressee, status)
  values (me, other, 'pending')
  on conflict (requester, addressee) do update
    set status = 'pending', created_at = now(), decided_at = null;
  return 'outgoing';
end;
$$;

create or replace function respond_friend_request(p_username text, p_pin text, p_from text, p_accept boolean)
returns text
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; other text; touched int;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  select s.username into other from students s where s.username_lower = lower(trim(p_from));
  if other is null then raise exception 'no_such_student'; end if;

  update friendships
  set status = case when p_accept then 'accepted' else 'declined' end,
      decided_at = now()
  where requester = other and addressee = me and status = 'pending';
  get diagnostics touched = row_count;
  if touched = 0 then raise exception 'no_such_request'; end if;

  return case when p_accept then 'friend' else 'declined' end;
end;
$$;

create or replace function remove_friend(p_username text, p_pin text, p_other text)
returns boolean
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; other text; touched int;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  select s.username into other from students s where s.username_lower = lower(trim(p_other));
  if other is null then raise exception 'no_such_student'; end if;

  delete from friendships
  where (requester = me and addressee = other)
     or (requester = other and addressee = me);
  get diagnostics touched = row_count;
  return touched > 0;
end;
$$;

create or replace function list_friends(p_username text, p_pin text)
returns table(username text, relation text, best_run_score int, since timestamptz)
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  return query
  select s.username,
         case
           when f.status = 'accepted' then 'friend'
           when f.requester = me then 'outgoing'
           else 'incoming'
         end,
         s.best_run_score,
         coalesce(f.decided_at, f.created_at)
  from friendships f
  join students s
    on s.username = case when f.requester = me then f.addressee else f.requester end
  where (f.requester = me or f.addressee = me)
    and f.status in ('accepted', 'pending')
  order by 2, 1;
end;
$$;

revoke execute on function friend_status(text, text) from public, anon;
grant execute on function find_students(text, text, text) to anon;
grant execute on function send_friend_request(text, text, text) to anon;
grant execute on function respond_friend_request(text, text, text, boolean) to anon;
grant execute on function remove_friend(text, text, text) to anon;
grant execute on function list_friends(text, text) to anon;
