-- Чья вещь: облик и своя аура питомца — его код; ауры стихий и звёзд — всем (null)
create or replace function pet_item_for(p_kind text, p_code text)
returns text language sql immutable as $$
  select case
    when p_kind = 'skin' then split_part(p_code, '_', 1)
    when p_kind = 'gear' then case
      when p_code in ('aura_leaves', 'aura_fireflies') then 'badger'
      when p_code in ('aura_moon', 'aura_feathers') then 'owl'
      when p_code in ('aura_tropics', 'aura_fruit') then 'toucan'
      when p_code in ('aura_rainbow', 'aura_butterflies') then 'cham'
      when p_code in ('aura_bubbles', 'aura_seabed') then 'crab'
      when p_code in ('aura_snow', 'aura_fish') then 'penguin'
      when p_code in ('aura_embers', 'aura_clouds') then 'dragon'
      when p_code in ('aura_coral', 'aura_jelly') then 'shrimp'
    end
  end;
$$;
