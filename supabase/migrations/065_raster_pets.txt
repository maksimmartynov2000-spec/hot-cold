-- Питомцы нарисованы картинками: вещи уходят, ауры и облики остаются.
-- Применять в Supabase: SQL Editor → вставить целиком → Run. Применить ДО
-- мёржа. Повторный запуск ничего не ломает и ничего не возвращает второй раз.
--
-- • Питомцы теперь — картинки (icons/pets, по шесть эмоций и шесть обликов у
--   каждого), а не схема из фигур. Вещи на такую картинку не надеть: корона,
--   шапочка учёного, очки-звёзды, цилиндр, шарф в ромбик, гирлянда, очки от
--   солнца, кепка, шляпа исследователя, спасательный круг, тельняшка, китель
--   и бинокль из игры уходят. Купившим — полная цена назад в монетах, как и
--   при переходе на шесть питомцев (064).
-- • Ауры остаются: четыре общие (пламя, иней, гроза, звёзды) и по две своих у
--   каждого питомца. Наряд питомца — аура и облик; прежние места (голова,
--   глаза, шея, тело) из нарядов снимаются.
-- • Наборы: питомец и две его ауры за монеты со скидкой 20 %.
-- • Товары дня с ушедшими вещами выбираются заново.
-- Всё считает база: цену, скидку и опыт из игры не прислать.

-- Чья вещь: облик и своя аура питомца — его код; ауры стихий и звёзд — всем (null)
create or replace function pet_item_for(p_kind text, p_code text)
returns text language sql immutable as $$
  select case
    when p_kind = 'skin' then split_part(p_code, '_', 1)
    when p_kind = 'gear' then case
      when p_code in ('aura_leaves', 'aura_fireflies') then 'badger'
      when p_code in ('aura_moon', 'aura_feathers') then 'owl'
      when p_code in ('aura_tropics', 'aura_fruit') then 'toucan'
      when p_code in ('aura_rainbow', 'aura_butterflies') then 'cham'
      when p_code in ('aura_bubbles', 'aura_seabed') then 'crab'
      when p_code in ('aura_snow', 'aura_fish') then 'penguin'
    end
  end;
$$;

-- Куда надевается: у питомца одно место для вещей — аура
create or replace function pet_gear_slot(p_code text)
returns text language sql immutable as $$
  select case when p_code like 'aura\_%' then 'aura' end;
$$;

-- ============ 3. Наряд: аура и облик ============
-- pet_gear у игрока: {"badger": {"aura": "aura_fire", "skin": "honey"}, "owl": {...}}.
-- Надеть на выбранного питомца ауру (aura: своя купленная или общая) или облик
-- (skin: свой купленный или star с 10-го уровня), снять — null.
-- Возвращает наряды всех питомцев
create or replace function set_pet_gear(p_username text, p_pin text, p_slot text, p_code text)
returns jsonb language plpgsql security definer set search_path = public, extensions, pg_temp as $$
declare me text; p text; g jsonb; o jsonb;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;
  select pet, coalesce(pet_gear, '{}'::jsonb) into p, g from students where username = me;
  if p is null or not cosmetic_has(me, 'pet', p) then raise exception 'no_pet'; end if;
  if p_slot is null or p_slot not in ('aura', 'skin') then raise exception 'bad_slot'; end if;
  if p_code is not null then
    if p_slot = 'skin' then
      if p_code = 'star' then
        if coalesce((select x.xp from pet_xp x where x.username = me and x.pet = p), 0) < pet_star_xp() then raise exception 'skin_locked'; end if;
      elsif not cosmetic_has(me, 'skin', p || '_' || p_code) then raise exception 'skin_locked'; end if;
    else
      if pet_gear_slot(p_code) is distinct from p_slot then raise exception 'bad_slot'; end if;
      if not cosmetic_has(me, 'gear', p_code) then raise exception 'gear_locked'; end if;
      if pet_item_for('gear', p_code) is not null and pet_item_for('gear', p_code) <> p then raise exception 'gear_locked'; end if;
    end if;
  end if;
  o := coalesce(g -> p, '{}'::jsonb);
  o := case when p_code is null then o - p_slot else o || jsonb_build_object(p_slot, p_code) end;
  g := g || jsonb_build_object(p, o);
  update students set pet_gear = g where username = me;
  return g;
end;
$$;

-- ============ 4. Вещи уходят: монеты назад ============
-- Купившим вещь — полная цена назад в монетах. Второй раз не вернётся:
-- запись о возврате с той же ссылкой в wallet_log вставляется только однажды,
-- а сами вещи из коллекций удаляются. Из нарядов снимаются прежние места
-- (голова, глаза, шея, тело); аура и облик остаются надетыми.
-- Возвращает, за сколько вещей вернулись монеты
create or replace function raster_pets_cleanup()
returns int language plpgsql security definer set search_path = public, pg_temp as $$
declare r record; n int; total int := 0;
begin
  for r in
    select c.username, c.code, o.coins
      from cosmetics c
      join (values ('crown', 450), ('mortar', 250), ('starglasses', 200), ('tophat', 250), ('argyle', 150), ('lei', 150), ('sunglasses', 200),
                   ('cap', 150), ('safari', 250), ('lifebuoy', 250), ('striped', 150), ('jacket', 300), ('binoculars', 200)) as o(code, coins)
        on c.kind = 'gear' and o.code = c.code
  loop
    insert into wallet(username) values (r.username) on conflict do nothing;
    insert into wallet_log(username, coins, gems, reason, ref, day)
    values (r.username, r.coins, 0, 'refund', 'gear:' || r.code, player_day(r.username))
    on conflict do nothing;
    get diagnostics n = row_count;
    if n > 0 then
      update wallet set coins = coins + r.coins, updated_at = now() where username = r.username;
      total := total + 1;
    end if;
    delete from cosmetics where username = r.username and kind = 'gear' and code = r.code;
  end loop;
  -- Наряды: {"badger": {"head": "crown", "aura": "aura_fire"}} → {"badger": {"aura": "aura_fire"}}
  update students s
     set pet_gear = (select coalesce(jsonb_object_agg(e.k, e.v - array['head', 'eyes', 'neck', 'body']), '{}'::jsonb)
                       from jsonb_each(s.pet_gear) as e(k, v) where jsonb_typeof(e.v) = 'object')
   where s.pet_gear is not null
     and exists (select 1 from jsonb_each(s.pet_gear) as e(k, v)
                  where jsonb_typeof(e.v) = 'object' and (e.v ? 'head' or e.v ? 'eyes' or e.v ? 'neck' or e.v ? 'body'));
  -- Товары дня с ушедшими вещами выбираются заново
  delete from shop_deals s
   where exists (select 1 from jsonb_array_elements(s.items) x
                  where not exists (select 1 from shop_catalog() c where c.kind = x->>'kind' and c.code = x->>'code'));
  return total;
end;
$$;
select raster_pets_cleanup();

-- ============ 5. Права ============
grant execute on function set_pet_gear(text, text, text, text) to anon;
revoke execute on function raster_pets_cleanup() from public, anon;
