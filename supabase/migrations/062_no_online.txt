-- Без онлайна со случайными соперниками. Применять в Supabase: SQL Editor →
-- вставить целиком → Run. Применить ДО мёржа, после 061.
--
-- Игра держится на трёх вещах: сам с ботами (Приключение, Тренировка,
-- Испытание), вдвоём на одном телефоне и с другом из «Друзей» на разных
-- телефонах.
-- • Удалено вместе с данными (вернуть нельзя): очередь онлайна («Играть»,
--   «Своя игра», режим дня), соперники-боты, рейтинг (звёзды, лиги, сезоны,
--   таблица лучших, рейтинговые партии), турнир выходных.
-- • Награды лиг, которые уже получены, остаются: иконки, фоны и рамки лиг
--   лежат в коллекции и надеваются, как награды пути.
-- • Партии с другом — только дружеские.
-- • Сезонный путь, сундук дня и серия дней — за любую игру: Приключение,
--   Испытание, Тренировку, игру вдвоём и с другом. За игры — не больше 150
--   очков пути в день, за задания дня и сундук — сверху.
-- • «Испытание выходных» вместо турнира: в субботу и воскресенье очко дают
--   заход Испытания на 5+ раундов и победа в Приключении на ★★★. 3 очка —
--   сундук и 1 💎, 5 — рамка месяца (или большой сундук) и 2 💎.
-- • Задания дня — без онлайна. Победы и игры в достижениях — любые
--   (Приключение, игры с другом), серия побед — тоже.
-- Миграцию можно запустить второй раз: ничего не сломается и не задвоится.

-- ============ 1. Награды лиг — в коллекцию, пока рейтинг ещё есть ============
do $$
declare codes text[] := array['bronze', 'silver', 'gold', 'platinum', 'diamond', 'legend'];
begin
  if to_regclass('public.ladder') is null then return; end if;
  -- Рамки лиг от серебра до высшей достигнутой (бронза была у всех)
  insert into cosmetics(username, kind, code, source)
  select l.username, 'frame', codes[i + 1], 'league'
  from ladder l
  join students s on s.username = l.username and not coalesce(s.is_bot, false)
  cross join lateral generate_series(1, least(5, greatest(0, l.peak))) i
  on conflict do nothing;
  -- Иконки и фоны лиг
  insert into cosmetics(username, kind, code, source)
  select l.username, r.kind, r.value, 'league'
  from ladder l
  join students s on s.username = l.username and not coalesce(s.is_bot, false)
  join league_rewards() r on r.league <= l.peak
  on conflict do nothing;
  -- Рамка «по лиге» становится своей: остаётся та, что была видна
  update students s set frame = codes[ladder_league(l.stars) + 1]
  from ladder l
  where l.username = s.username and s.frame is null and not coalesce(s.is_bot, false)
    and ladder_league(l.stars) >= 1;
end $$;

-- Ключ сезона пути — месяц по UTC (раньше — ladder_season)
create or replace function season_key()
returns text language sql stable as $$ select to_char(now() at time zone 'UTC', 'YYYY-MM'); $$;

-- Сколько очков пути в день можно получить за игры
create or replace function play_xp_cap() returns int language sql immutable as $$ select 150; $$;

-- Сыграна игра: день отмечен (сундук дня открыт), очки пути — не больше
-- play_xp_cap() в день за игры. Возвращает, сколько очков начислено
create or replace function play_done(p_username text, p_xp int)
returns int
language plpgsql security definer set search_path = public, pg_temp as $$
declare d date; have int; amt int;
begin
  if not wallet_person(p_username) then return 0; end if;
  d := player_day(p_username);
  insert into online_days(username, day, played) values (p_username, d, true)
  on conflict (username, day) do update set played = true;
  select play_xp into have from online_days where username = p_username and day = d for update;
  amt := least(greatest(coalesce(p_xp, 0), 0), greatest(0, play_xp_cap() - have));
  if amt > 0 then
    update online_days set play_xp = play_xp + amt where username = p_username and day = d;
    perform season_xp(p_username, amt);
  end if;
  return amt;
end;
$$;

-- Игра в статистику достижений: игры, победы, серия побед подряд
create or replace function stats_game(p_username text, p_won boolean)
returns void
language plpgsql security definer set search_path = public, pg_temp as $$
begin
  if not wallet_person(p_username) then return; end if;
  insert into player_stats(username) values (p_username) on conflict do nothing;
  update player_stats
     set online_games = online_games + 1,
         online_wins = online_wins + case when p_won then 1 else 0 end,
         win_streak = case when p_won then win_streak + 1 else 0 end,
         best_win_streak = greatest(best_win_streak, case when p_won then win_streak + 1 else 0 end)
   where username = p_username;
end;
$$;

-- Игра без интернета кончилась: Тренировка, игра вдвоём на одном телефоне,
-- проигранный бой Приключения (победу присылает adventure_result). День
-- отмечен, очки пути, задания «сыграйте»
create or replace function local_game_done(p_username text, p_pin text, p_kind text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; d date; last timestamptz; xp int;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;
  if p_kind is null or p_kind not in ('solo', 'duel', 'adv') then raise exception 'bad_kind'; end if;
  d := player_day(me);
  insert into online_days(username, day) values (me, d) on conflict do nothing;
  select local_at into last from online_days where username = me and day = d for update;
  -- Не чаще раза в 5 секунд: игра короче не бывает
  if last is not null and last > now() - interval '5 seconds' then raise exception 'too_fast'; end if;
  update online_days set local_at = now() where username = me and day = d;
  if p_kind = 'adv' then perform stats_game(me, false); end if;
  perform quest_event(me, 'play', 1);
  xp := play_done(me, 10);
  perform achievements_sync(me);
  return jsonb_build_object('xp', xp);
end;
$$;

-- ============ Переписано: партия, профиль, достижения — без рейтинга и ботов ============
-- Ход в партии: рейтинга больше нет — конец партии без пересчёта
create or replace function match_guess_core(p_match_id bigint, p_seat int, p_guess int)
returns int
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare m matches; seat int; dist int; band int; used_double boolean;
        hit record; btype text; radius int; done boolean; reflected text;
begin
  seat := p_seat;
  select * into m from matches where id = p_match_id for update;
  if m.id is null then raise exception 'no_such_match'; end if;
  if m.status <> 'active' then raise exception 'not_playing'; end if;
  if m.round_over or m.match_over then raise exception 'round_over'; end if;
  if m.cur <> seat then raise exception 'not_your_turn'; end if;
  -- Пока оба не пришли и не отсчитано 3-2-1, ходить нельзя
  if m.lobby or (m.start_at is not null and m.start_at > now()) then
    raise exception 'not_started';
  end if;
  if m.skip_turn[seat + 1] or m.auto_lava[seat + 1] then raise exception 'forced_turn'; end if;
  if p_guess is null or p_guess < m.range_min or p_guess > m.range_max then
    raise exception 'out_of_range';
  end if;

  dist := abs(p_guess - m.secret);
  band := feedback_tier(match_span(m), dist);
  -- Для разбора: ход сделан под помехой — игрок видел не все ходы
  insert into match_moves(match_id, round, seat, guess, tier, limited)
  values (p_match_id, m.round, seat, p_guess, band,
          m.fog[seat + 1] or m.blind[seat + 1] or m.short_memory[seat + 1]);

  radius := bonus_near_radius(match_span(m));
  select * into hit from match_bonuses b
  where b.match_id = p_match_id and b.round = m.round and not b.taken and b.value = p_guess;

  btype := null;
  if hit.type is not null then
    btype := hit.type;
    if match_bonus_blocked(m, btype, 1 - seat) then btype := 'extra'; end if;
    update match_bonuses set taken = true
    where match_id = p_match_id and round = m.round and value = p_guess;
  end if;

  -- Зеркало (миграция 060) отражает ловушку в соперника, щит (057) просто
  -- отбивает. Зеркало — первым: оно сильнее, а щит останется на потом
  update matches set shield_trap = null where id = p_match_id;
  reflected := null;
  if btype in ('skip', 'gift', 'memory') and m.mirror[seat + 1] then
    update matches set mirror[seat + 1] = false, shield_trap = btype where id = p_match_id;
    reflected := btype;
    btype := 'mirrorUsed';
  elsif btype in ('skip', 'gift', 'memory') and m.shield[seat + 1] then
    update matches set shield[seat + 1] = false, shield_trap = btype where id = p_match_id;
    btype := 'shieldUsed';
  end if;

  -- Сходил — значит на месте: счётчик молчания обнуляется
  update matches
  set last_bonus = btype,
      last_bonus_by = seat,
      last_timeout = false,
      timeouts[seat + 1] = 0,
      near_bonus = (btype is null and exists (
        select 1 from match_bonuses b
        where b.match_id = p_match_id and b.round = m.round and not b.taken
          and abs(b.value - p_guess) <= radius))
  where id = p_match_id;

  if reflected is not null then
    perform apply_match_bonus(p_match_id, reflected, 1 - seat, p_guess);
  elsif btype is not null and btype <> 'shieldUsed' then
    perform apply_match_bonus(p_match_id, btype, seat, p_guess);
  end if;

  select * into m from matches where id = p_match_id;
  used_double := m.armed;

  if dist = 0 then
    done := (m.wins[seat + 1] + 1) >= m.wins_needed;
    update matches
    set round_over = true, round_winner = seat,
        wins[seat + 1] = m.wins[seat + 1] + 1,
        match_over = done,
        status = case when done then 'finished' else status end,
        armed = false, turn_deadline = null, updated_at = now()
    where id = p_match_id;
  elsif used_double then
    update matches set armed = false, updated_at = now() where id = p_match_id;
    select * into m from matches where id = p_match_id;
    update matches set turn_deadline = now() + make_interval(secs => turn_seconds(m))
    where id = p_match_id;
  else
    perform pass_turn(p_match_id, false);
  end if;

  return band;
end;
$$;

-- Время хода вышло: без ботов и рейтинга
create or replace function apply_turn_timeout(p_match_id bigint)
returns boolean
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare m matches; gone int; rival int;
begin
  select * into m from matches where id = p_match_id for update;
  if m.id is null then return false; end if;
  if m.status <> 'active' or m.round_over or m.match_over then return false; end if;
  if m.turn_deadline is null or m.turn_deadline > now() then return false; end if;

  insert into match_moves(match_id, round, seat, guess, tier, kind)
  values (p_match_id, m.round, m.cur, null, null, 'timeout');

  gone := m.timeouts[m.cur + 1] + 1;
  rival := 1 - m.cur;

  update matches
  set skip_turn[m.cur + 1] = false,
      auto_lava[m.cur + 1] = false,
      near_bonus = false,
      last_bonus = null,
      timeouts[m.cur + 1] = gone
  where id = p_match_id;

  if gone >= (case when m.variant = 'blitz' then 3 else 2 end) then
    update matches
    set round_over = true,
        round_winner = rival,
        match_over = true,
        status = 'finished',
        forfeit_by = m.cur,
        turn_deadline = null,
        updated_at = now()
    where id = p_match_id;
    return true;
  end if;

  perform pass_turn(p_match_id, true);
  return true;
end;
$$;

-- Выйти из партии с другом: партия кончается без результата
create or replace function leave_match(p_username text, p_pin text, p_match_id bigint)
returns boolean
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; m matches; my_seat int;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  select * into m from matches where id = p_match_id for update;
  if m.id is null then raise exception 'no_such_match'; end if;
  my_seat := match_seat(m, me);
  if my_seat is null then raise exception 'not_your_match'; end if;
  if m.status not in ('invited', 'active') then return false; end if;

  -- Партия ещё не началась — оба не пришли. Уход её отменяет, без результата
  if m.status = 'active' and m.lobby then
    update matches set status = 'expired', turn_deadline = null, updated_at = now()
    where id = p_match_id;
    return true;
  end if;

  update matches set status = 'finished', match_over = true, updated_at = now()
  where id = p_match_id;
  return true;
end;
$$;

-- Удалить аккаунт: без рейтинга; монеты и кристаллы — тоже
create or replace function delete_account(p_username text, p_pin text)
returns text
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; gone text; m matches;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  for m in select * from matches
           where status in ('invited', 'active') and (p0 = me or p1 = me) loop
    update matches set status = 'finished', match_over = true, updated_at = now()
    where id = m.id;
  end loop;

  delete from friendships where requester = me or addressee = me;
  delete from run_sessions where username = me;
  delete from friend_chat where a = me or b = me;
  delete from friend_chat_read where username = me or other = me;
  delete from push_subscriptions where username = me;
  -- И то, что ждало отправки ему, и то, что ждало отправки про него: имя в
  -- колонке who внешним ключом не связано и пережило бы переименование
  delete from push_outbox where username = me or who = me;
  delete from runs where username = me;
  delete from wallet_log where username = me;
  delete from wallet where username = me;

  gone := '#' || nextval('deleted_player_seq');
  update students
  set username = gone,
      pin_hash = crypt(gen_random_uuid()::text, gen_salt('bf')),
      pin_hint = null,
      best_run_score = 0,
      best_run_rounds = 0,
      failed_logins = 0,
      locked_until = null
  where username = me;

  return gone;
end;
$$;

-- Сменить имя: таблицы соперничества (для рейтинга) больше нет
create or replace function rename_student(p_username text, p_pin text, p_new text)
returns text
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; nn text; last_rename timestamptz; chat friend_chat[];
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  nn := trim(coalesce(p_new, ''));
  perform check_new_username(nn);
  if nn = me then return me; end if;

  select renamed_at into last_rename from students where username = me;
  if last_rename is not null and last_rename > now() - interval '24 hours' then
    raise exception 'rename_too_soon:%',
      greatest(1, ceil(extract(epoch from (last_rename + interval '24 hours' - now())) / 3600))::int;
  end if;

  if exists (select 1 from students where username_lower = lower(nn) and username <> me) then
    raise exception 'name_taken';
  end if;

  select coalesce(array_agg(c), '{}') into chat from friend_chat c where c.a = me or c.b = me;
  delete from friend_chat where a = me or b = me;

  update students set username = nn, renamed_at = now() where username = me;

  update push_outbox set who = nn where who = me;
  update runs set username = nn where username = me;

  perform set_config('hotcold.quiet_chat', 'on', true);
  insert into friend_chat(id, a, b, sender, code, body, created_at) overriding system value
  select c.id, least(x.a2, x.b2), greatest(x.a2, x.b2),
         case when c.sender = me then nn else c.sender end, c.code, c.body, c.created_at
  from unnest(chat) c,
       lateral (select case when c.a = me then nn else c.a end as a2,
                       case when c.b = me then nn else c.b end as b2) x;
  perform set_config('hotcold.quiet_chat', 'off', true);

  return nn;
end;
$$;

-- Состояние партии: без ботов, рейтинга и звёзд
create or replace function match_state(p_username text, p_pin text, p_match_id bigint)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; m matches; my_seat int; moves jsonb; reveal boolean;
        visible int; hide_mine boolean; hide_theirs boolean; chat jsonb;
        other text; rm matches; rem jsonb;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  perform expire_stale_matches();

  select * into m from matches where id = p_match_id;
  if m.id is null then raise exception 'no_such_match'; end if;
  my_seat := match_seat(m, me);
  if my_seat is null then raise exception 'not_your_match'; end if;

  perform match_flow_tick(p_match_id, my_seat);

  perform apply_turn_timeout(p_match_id);

  select * into m from matches where id = p_match_id;

  reveal := m.round_over or m.match_over or m.status <> 'active';
  hide_mine := (not reveal) and m.blind[my_seat + 1];
  hide_theirs := (not reveal) and m.fog[my_seat + 1];
  visible := case when (not reveal) and m.short_memory[my_seat + 1] then 2 else 4 end;

  select coalesce(jsonb_agg(x order by x_id), '[]'::jsonb) into moves from (
    select mv.id as x_id,
           case
             when mv.kind = 'timeout'
               then jsonb_build_object('seat', mv.seat, 'timeout', true)
             when (mv.seat = my_seat and hide_mine and mv.id > m.blind_from[my_seat + 1])
               or (mv.seat <> my_seat and hide_theirs and mv.id > m.fog_from[my_seat + 1])
               then jsonb_build_object('seat', mv.seat, 'hidden', true)
             else jsonb_build_object('seat', mv.seat, 'guess', mv.guess, 'tier', mv.tier)
           end as x
    from (select * from match_moves mv2
          where mv2.match_id = p_match_id and mv2.round = m.round
          order by mv2.id desc limit visible) mv
  ) q;

  select coalesce(jsonb_agg(jsonb_build_object(
           'id', c.id, 'seat', c.seat, 'code', c.code,
           'ago', floor(extract(epoch from (now() - c.created_at)))::int) order by c.id), '[]'::jsonb)
    into chat
  from (select * from match_chat c2 where c2.match_id = p_match_id
        order by c2.id desc limit 6) c;

  -- Реванш после этой партии: приглашение или уже идущая
  if m.status = 'finished' then
    other := case when m.p0 = me then m.p1 else m.p0 end;
    select * into rm from matches y
    where y.id > m.id and y.status in ('invited', 'active', 'declined')
      and ((y.p0 = me and y.p1 = other) or (y.p0 = other and y.p1 = me))
    order by y.id desc limit 1;
    if rm.id is not null then
      rem := jsonb_build_object('id', rm.id, 'status', rm.status, 'mine', rm.p0 = me);
    end if;
  end if;

  return jsonb_build_object(
    'id', m.id, 'seat', my_seat,
    'names', jsonb_build_array(m.p0, m.p1),
    'status', m.status, 'winsNeeded', m.wins_needed,
    'rangeMin', m.range_min, 'rangeMax', m.range_max, 'frost', m.frost,
    'round', m.round, 'wins', m.wins, 'cur', m.cur, 'tokens', m.tokens, 'armed', m.armed,
    'roundOver', m.round_over, 'roundWinner', m.round_winner, 'matchOver', m.match_over,
    'moves', moves,
    'secret', case when reveal then to_jsonb(m.secret) else 'null'::jsonb end,
    'bonusesOn', m.bonuses_on,
    'fog', m.fog, 'blind', m.blind, 'autoLava', m.auto_lava,
    'skip', m.skip_turn, 'shortMemory', m.short_memory, 'rush', m.rush,
    'nearBonus', m.near_bonus, 'lastBonus', m.last_bonus, 'lastBonusBy', m.last_bonus_by,
    'lastTimeout', m.last_timeout,
    'forced', forced_turn_kind(m),
    'turnSeconds', turn_seconds(m),
    'secondsLeft', case when m.turn_deadline is null then null
                        else greatest(0, ceil(extract(epoch from (m.turn_deadline - now()))))::int end,
    'chat', chat,
    'forfeitBy', m.forfeit_by,
    'updatedAt', m.updated_at,
    'lobby', m.lobby,
    'present', jsonb_build_array(match_side_here(m, 0), match_side_here(m, 1)),
    'lobbyLeft', case when m.lobby and m.accepted_at is not null
                      then greatest(0, ceil(extract(epoch from (m.accepted_at + interval '5 minutes' - now()))))::int end,
    'startsIn', case when m.start_at is not null and m.start_at > now()
                     then ceil(extract(epoch from (m.start_at - now())))::int end,
    'nextIn', case when m.status = 'active' and m.round_over and not m.match_over and m.round_ended_at is not null
                   then greatest(0, ceil(extract(epoch from (m.round_ended_at + interval '10 seconds' - now()))))::int end,
    'ready', jsonb_build_array(match_side_ready(m, 0), match_side_ready(m, 1)),
    'rematch', rem
  )
  -- Бонусы из Приключения (миграция 057): щит видят оба, ответ Рентгена и
  -- Половины — только тот, кто взял. Отдельным объектом: у jsonb_build_object
  -- предел — сто аргументов
  || jsonb_build_object(
    'shield', m.shield, 'frozen', m.frozen, 'shieldTrap', m.shield_trap,
    -- Новые бонусы (миграция 060): зеркало видят оба, знание — только взявший
    'mirror', m.mirror,
    'know', jsonb_build_object('parity', m.know_parity[my_seat + 1], 'half', m.know_half[my_seat + 1],
                               'div3', m.know_div3[my_seat + 1], 'div5', m.know_div5[my_seat + 1],
                               'gt', m.know_gt[my_seat + 1], 'lt', m.know_lt[my_seat + 1]));
end;
$$;

-- Разбор партии: без звёзд рейтинга
create or replace function match_review(p_username text, p_pin text, p_match_id bigint)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; m matches; moves jsonb;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  select * into m from matches where id = p_match_id;
  if m.id is null then raise exception 'no_such_match'; end if;
  if match_seat(m, me) is null then raise exception 'not_your_match'; end if;
  -- Пока партия идёт, ходы прошлых раундов — подсказка о том, как играет
  -- соперник, а текущего — прямая подсказка. Только после конца
  if not m.match_over then raise exception 'match_not_over'; end if;

  select coalesce(jsonb_agg(jsonb_build_object(
           'round', mv.round, 'seat', mv.seat, 'guess', mv.guess, 'tier', mv.tier,
           'timeout', mv.kind = 'timeout', 'forced', mv.forced, 'limited', mv.limited)
         order by mv.id), '[]'::jsonb)
    into moves
  from match_moves mv where mv.match_id = p_match_id;

  return jsonb_build_object('id', m.id, 'moves', moves);
end;
$$;

-- Сбор перед партией: только люди
create or replace function matches_flow()
returns trigger
language plpgsql
set search_path = public, pg_temp
as $$
begin
  if tg_op = 'INSERT' then
    if new.status = 'active' then
      new.lobby := true;
      new.accepted_at := now();
    end if;
  else
    if old.status = 'invited' and new.status = 'active' then
      new.lobby := true;
      new.accepted_at := now();
    end if;
    if new.round_over and not coalesce(old.round_over, false) then
      new.round_ended_at := now();
    end if;
    if not new.round_over and coalesce(old.round_over, false) then
      new.ready := array[false, false];
    end if;
  end if;
  if new.lobby then new.turn_deadline := null; end if;
  return new;
end;
$$;

-- Перерыв между раундами: без ботов
create or replace function match_flow_tick(p_match_id bigint, p_seat int)
returns void
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare m matches;
begin
  update matches set seen_at[p_seat + 1] = now() where id = p_match_id;
  select * into m from matches where id = p_match_id for update;
  if m.id is null or m.status <> 'active' then return; end if;

  if m.lobby then
    if match_side_here(m, 0) and match_side_here(m, 1) then
      update matches
      set lobby = false,
          start_at = now() + interval '3 seconds',
          turn_deadline = now() + interval '3 seconds' + make_interval(secs => turn_seconds(m))
      where id = p_match_id;
    end if;
    return;
  end if;

  if m.round_over and not m.match_over and m.round_ended_at <= now() - interval '10 seconds' then
    perform match_start_next(p_match_id);
  end if;
end;
$$;

set search_path = public, pg_temp

-- Реванш с другом: та же партия, дружеская
create or replace function rematch(p_username text, p_pin text, p_match_id bigint)
returns bigint
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; m matches; other text; live int; new_id bigint; x matches;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  perform expire_stale_matches();

  select * into m from matches where id = p_match_id;
  if m.id is null then raise exception 'no_such_match'; end if;
  if match_seat(m, me) is null then raise exception 'not_your_match'; end if;
  if m.status in ('invited', 'active') then raise exception 'match_already_live'; end if;

  other := case when m.p0 = me then m.p1 else m.p0 end;
  if other like '#%' then raise exception 'no_such_student'; end if;

  -- Соперник уже позвал на реванш — принять его вызов
  select * into x from matches
  where id > m.id and status = 'invited' and p0 = other and p1 = me
  order by id desc limit 1 for update;
  if x.id is not null then
    update matches set status = 'active', starter = 0, updated_at = now() where id = x.id;
    perform start_match_round(x.id);
    return x.id;
  end if;

  select count(*) into live from matches
  where status in ('invited', 'active')
    and ((p0 = me and p1 = other) or (p0 = other and p1 = me));
  if live > 0 then raise exception 'match_already_live'; end if;

  insert into matches(p0, p1, wins_needed, range_min, range_max, frost, bonuses_on, bonus_count, status)
  values (me, other, m.wins_needed, m.range_min, m.range_max, m.frost, m.bonuses_on, m.bonus_count, 'invited')
  returning id into new_id;
  return new_id;
end;
$$;

-- Партия с исходом: игры, победы и серия — в статистику, задания «сыграйте»
-- и «победите», достижения
create or replace function matches_progress()
returns trigger
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare seat int; who text; w int;
begin
  w := match_winner(new);
  if w is null then return null; end if;
  for seat in 0..1 loop
    who := case when seat = 0 then new.p0 else new.p1 end;
    if who like '#%' then continue; end if;
    perform stats_game(who, w = seat);
    perform quest_event(who, 'play', 1);
    if w = seat then perform quest_event(who, 'win', 1); end if;
    perform achievements_sync(who);
  end loop;
  return null;
end;
$$;

-- Партия с другом: день отмечен, очки пути (10, за победу — 20) и монеты
-- (победа 15, поражение 5, сдавшемуся — 0; за партии и Испытание — не
-- больше wallet_cap() в день)
create or replace function matches_pass()
returns trigger
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare seat int; who text; w int;
begin
  w := match_winner(new);
  if w is null then return null; end if;
  for seat in 0..1 loop
    who := case when seat = 0 then new.p0 else new.p1 end;
    if who like '#%' then continue; end if;
    perform play_done(who, 10 + case when w = seat then 10 else 0 end);
    perform wallet_earn(who, case when w = seat then 15 when new.forfeit_by = seat then 0 else 5 end,
                        'online', 'match:' || new.id);
  end loop;
  return null;
end;
$$;

-- ============ 8. Задания дня без онлайна ============
create or replace function quest_pool()
returns table(slot int, ord int, code text, goal int)
language sql
immutable
as $$
  values (1, 1, 'play2', 2), (1, 2, 'play3', 3), (1, 3, 'adv_win2', 2),
         (2, 1, 'win1', 1), (2, 2, 'win2', 2),
         (3, 1, 'run1', 1), (3, 2, 'run_rounds5', 5), (3, 3, 'run_best3', 3), (3, 4, 'adv_stars3', 3);
$$;

-- play — любая игра (Приключение, Испытание, Тренировка, вдвоём, с другом);
-- win — победа в Приключении или над другом
create or replace function quest_event(p_username text, p_event text, p_n int)
returns void
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare d date; was_done boolean; now_done boolean;
begin
  if p_username is null or p_username like '#%' or p_username like '@bot:%' then return; end if;
  d := quest_ensure(p_username);
  select bool_and(progress >= goal) into was_done from daily_quests where username = p_username and day = d;
  update daily_quests q
  set progress = least(q.goal, case
      when q.code in ('play2', 'play3') and p_event = 'play' then q.progress + 1
      when q.code in ('win1', 'win2') and p_event = 'win' then q.progress + 1
      when q.code = 'run1' and p_event = 'run' and p_n > 0 then q.progress + 1
      when q.code = 'run_rounds5' and p_event = 'run' then q.progress + greatest(0, p_n)
      when q.code = 'run_best3' and p_event = 'run' then greatest(q.progress, p_n)
      -- Приключение: победа и новые звёзды
      when q.code = 'adv_win2' and p_event = 'adv_win' then q.progress + 1
      when q.code = 'adv_stars3' and p_event = 'adv_star' then q.progress + greatest(0, p_n)
      else q.progress end)
  where q.username = p_username and q.day = d;
  select bool_and(progress >= goal) into now_done from daily_quests where username = p_username and day = d;
  if now_done and not was_done then
    insert into player_stats(username) values (p_username) on conflict do nothing;
    update player_stats
    set quest_streak = case when quest_last = d - 1 then quest_streak + 1
                            when quest_last = d then quest_streak else 1 end,
        quest_last = d
    where username = p_username;
    update player_stats set quest_best = greatest(quest_best, quest_streak) where username = p_username;
  end if;
end;
$$;

-- Задания, которых больше нет: на рейтинг и режим дня — заменить равными
update daily_quests
   set code = case code when 'daily_win1' then 'win1' else 'play2' end,
       goal = case code when 'daily_win1' then 1 else 2 end
 where code in ('ladder2', 'daily2', 'daily_win1');

-- ============ 10. Профиль, задания, путь ============
create or replace function my_progress(p_username text, p_pin text, p_tz text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; s students; d date; st player_stats; quests jsonb; ach jsonb; streak int; fr int; peak int;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;
  select * into s from students where username = me;
  if p_tz is not null and p_tz is distinct from s.tz
     and (s.tz is null or s.tz_set_at is null or s.tz_set_at <= now() - interval '20 hours')
     and exists (select 1 from pg_timezone_names where name = p_tz) then
    update students set tz = p_tz, tz_set_at = now() where username = me;
  end if;
  perform achievements_sync(me);
  d := quest_ensure(me);
  select * into st from player_stats where username = me;
  select jsonb_agg(jsonb_build_object('code', q.code, 'goal', q.goal, 'progress', q.progress) order by q.slot)
    into quests
  from daily_quests q where q.username = me and q.day = d;
  select coalesce(jsonb_agg(jsonb_build_object('code', a.code, 'at', a.at) order by a.at), '[]'::jsonb)
    into ach
  from achievements a where a.username = me;
  select count(*) into fr from friendships
  where status = 'accepted' and me in (requester, addressee);
  -- Лига — только та, что уже была достигнута: рейтинга больше нет
  select coalesce(max(case a.code when 'league_silver' then 1 when 'league_gold' then 2 when 'league_platinum' then 3
                                  when 'league_diamond' then 4 when 'legend' then 5 end), 0)
    into peak from achievements a where a.username = me;
  streak := case when st.quest_last >= d - 1 then st.quest_streak else 0 end;
  return jsonb_build_object(
    'day', d,
    'quests', coalesce(quests, '[]'::jsonb),
    'streak', coalesce(streak, 0),
    'bestStreak', coalesce(st.quest_best, 0),
    'doneToday', coalesce(st.quest_last = d, false),
    'secondsLeft', greatest(0, floor(extract(epoch from
        ((d + 1)::timestamp at time zone coalesce((select tz from students where username = me), 'UTC')) - now())))::int,
    'achievements', ach,
    'stats', jsonb_build_object(
      'wins', coalesce(st.online_wins, 0),
      'games', coalesce(st.online_games, 0),
      'winStreak', coalesce(st.best_win_streak, 0),
      'firstTry', ach_first_try(me),
      'modes', ach_modes(me),
      'peak', peak,
      'questBest', coalesce(st.quest_best, 0),
      'runBest', coalesce(st.run_best, 0),
      'friends', fr,
      'rivals', ach_friends_beaten(me),
      'tutorial', coalesce(s.tutorial_done, false),
      'advStars', adventure_star_total(s.adventure_stars),
      'advWorlds', adventure_worlds(s.adventure),
      'advGold', adventure_gold_worlds(s.adventure_stars)));
end;
$$;

-- Путь: месяц — по season_key
create or replace function season_xp(p_username text, p_xp int)
returns int
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare se text; before int; after int; r record;
begin
  if p_username is null or p_username like '#%' or p_username like '@bot:%' or coalesce(p_xp, 0) <= 0 then return 0; end if;
  se := season_key();
  insert into season_pass(username, season) values (p_username, se) on conflict do nothing;
  select xp into before from season_pass where username = p_username and season = se for update;
  update season_pass set xp = before + p_xp where username = p_username and season = se returning xp into after;
  for r in select * from season_rewards() s
           where s.level > least(pass_max_level(), before / pass_level_xp())
             and s.level <= least(pass_max_level(), after / pass_level_xp()) loop
    perform cosmetic_grant(p_username, r.kind, r.code, 'pass:' || se);
  end loop;
  return after;
end;
$$;

-- Сундук, серия, путь, своё и выходные одним запросом
create or replace function my_online(p_username text, p_pin text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; d date; od online_days; st online_streak; se text; xp int; s students; owned jsonb; rew jsonb;
        wg weekend_goal; utc timestamp; sat timestamp;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;
  d := player_day(me);
  select * into od from online_days where username = me and day = d;
  select * into st from online_streak where username = me;
  se := season_key();
  select coalesce(p.xp, 0) into xp from (select 1) x left join season_pass p on p.username = me and p.season = se;
  select * into s from students where username = me;
  select coalesce(jsonb_agg(jsonb_build_object('kind', c.kind, 'code', c.code) order by c.got_at), '[]'::jsonb)
    into owned from cosmetics c where c.username = me;
  select jsonb_agg(jsonb_build_object('level', r.level, 'kind', r.kind, 'code', r.code) order by r.level, r.kind)
    into rew from season_rewards() r;
  select * into wg from weekend_goal w where w.username = me and w.week = weekend_row_key();
  utc := hc_now() at time zone 'UTC';
  -- Ближайшая суббота 00:00 UTC (на выходных — та, что идёт)
  sat := date_trunc('week', utc) + interval '5 days';
  return jsonb_build_object(
    'day', d,
    'played', coalesce(od.played, false),
    'chestOpened', od.chest_at is not null,
    'chest', od.chest,
    'streak', case when st.last_day >= d - 1 then st.streak
                   when st.last_day = d - 2 and coalesce(st.freeze_week, '') <> to_char(d, 'IYYY-IW') then st.streak
                   else 0 end,
    'bestStreak', coalesce(st.best, 0),
    'freezeLeft', coalesce(st.freeze_week, '') <> to_char(d, 'IYYY-IW'),
    'season', se,
    'xp', xp,
    'levelXp', pass_level_xp(),
    'maxLevel', pass_max_level(),
    'playXp', coalesce(od.play_xp, 0),
    'playXpCap', play_xp_cap(),
    'seasonLeft', greatest(0, floor(extract(epoch from
        (date_trunc('month', now() at time zone 'UTC') + interval '1 month') - (now() at time zone 'UTC'))))::int,
    'rewards', rew,
    'streakRewards', (select jsonb_agg(jsonb_build_object('days', r.days, 'kind', r.kind, 'code', r.code) order by r.days) from streak_rewards() r),
    'owned', owned,
    'frame', s.frame,
    'title', s.title,
    'weekend', jsonb_build_object(
      'on', weekend_on(),
      'points', coalesce(wg.points, 0),
      'claim3', wg.claim3,
      'claim5', wg.claim5,
      'frame', cup_frame_code(),
      'startsIn', case when weekend_on() then 0 else floor(extract(epoch from (sat - utc)))::int end,
      'endsIn', case when weekend_on() then floor(extract(epoch from (sat + interval '2 days' - utc)))::int else 0 end));
end;
$$;

-- Награды «Испытания выходных»: 3 очка — сундук и 1 💎, 5 — рамка месяца
-- (у кого она есть — большой сундук) и 2 💎. Забрать можно до следующих выходных
create or replace function weekend_claim(p_username text, p_pin text, p_goal int)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; c weekend_goal; xp int; sur record; got jsonb := null; frame boolean := false; res jsonb; gems int;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;
  if p_goal not in (3, 5) then raise exception 'bad_goal'; end if;
  select * into c from weekend_goal where username = me and week = weekend_row_key() for update;
  if c.username is null or c.points < p_goal then raise exception 'cup_locked'; end if;
  if p_goal = 3 and c.claim3 is not null then return c.claim3 || jsonb_build_object('already', true); end if;
  if p_goal = 5 and c.claim5 is not null then return c.claim5 || jsonb_build_object('already', true); end if;

  if p_goal = 5 and cosmetic_grant(me, 'frame', cup_frame_code(), 'weekend:' || c.week) then
    frame := true;
    xp := 50;
  else
    xp := case when p_goal = 3 then 50 else 100 end;
    -- Сундук за 3 очка — сюрприз всегда, если ещё есть что дать; большой — через раз
    if p_goal = 3 or random() < 0.5 then
      select x.* into sur from chest_surprises() x where not cosmetic_has(me, x.kind, x.code) order by random() limit 1;
      if sur.kind is not null and cosmetic_grant(me, sur.kind, sur.code, 'weekend:' || c.week) then
        got := jsonb_build_object('kind', sur.kind, 'code', sur.code);
      end if;
    end if;
  end if;
  perform season_xp(me, xp);
  gems := case when p_goal = 3 then 1 else 2 end;
  if not wallet_add(me, 0, gems, 'weekend', 'weekend:' || c.week || ':' || p_goal) then gems := 0; end if;
  res := jsonb_build_object('goal', p_goal, 'xp', xp, 'surprise', got,
                            'frame', case when frame then cup_frame_code() end, 'gems', gems);
  if p_goal = 3 then update weekend_goal set claim3 = res where username = me and week = c.week;
  else update weekend_goal set claim5 = res where username = me and week = c.week; end if;
  return res;
end;
$$;

-- Приключение: бой выигран — звёзды, монеты и кристаллы, а ещё: игра в
-- статистику, день и очки пути (10 и по 5 за новую звезду), очко выходных
-- за ★★★
create or replace function adventure_result(p_username text, p_pin text, p_n int, p_stars int)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; s students; was int; got int; st text; pay jsonb;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;
  if p_n is null or p_n < 0 or p_n > 499 or p_stars is null or p_stars < 1 or p_stars > 3 then
    raise exception 'bad_result';
  end if;
  select * into s from students where username = me for update;
  -- Дальше открытого соперника не прыгнуть
  if p_n > s.adventure then raise exception 'not_open'; end if;
  st := rpad(s.adventure_stars, greatest(length(s.adventure_stars), p_n + 1), '0');
  was := substr(st, p_n + 1, 1)::int;
  got := greatest(0, p_stars - was);
  if got > 0 then st := overlay(st placing p_stars::text from p_n + 1 for 1); end if;
  update students
     set adventure_stars = st,
         adventure = greatest(adventure, p_n + 1)
   where username = me
  returning * into s;
  perform stats_game(me, true);
  perform quest_event(me, 'adv_win', 1);
  perform quest_event(me, 'play', 1);
  perform quest_event(me, 'win', 1);
  if got > 0 then perform quest_event(me, 'adv_star', got); end if;
  perform play_done(me, 10 + 5 * got);
  if p_stars = 3 then perform weekend_point(me); end if;
  perform achievements_sync(me);
  pay := adventure_wallet(me);
  return jsonb_build_object('done', s.adventure, 'stars', s.adventure_stars,
                            'total', adventure_star_total(s.adventure_stars), 'new', got,
                            'coins', pay->'coins', 'gems', pay->'gems');
end;
$$;

-- Испытание: итог захода — как раньше, монеты, день и очки пути (15),
-- очко выходных за 5+ раундов
create or replace function run_report(p_username text, p_pin text, p_rounds int, p_score int)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; r run_sessions; best int; best_rounds int; coins int;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;
  if p_rounds is null or p_rounds < 0 or p_rounds > 60 then raise exception 'bad_result'; end if;
  -- Больше, чем за 60 раундов лучшей игры, набрать нельзя: очки за раунд n —
  -- не больше 100·n·14, с риском и подарком ×4
  if p_score is null or p_score < 0 or p_score > 5600 * p_rounds * (p_rounds + 1) / 2 + 1 then
    raise exception 'bad_result';
  end if;
  select * into r from run_sessions where username = me;
  -- Не чаще раза в 5 секунд: итог присылается один раз на заход
  if r.username is not null and r.status = 'finished' and r.updated_at > now() - interval '5 seconds' then
    raise exception 'too_fast';
  end if;
  insert into run_sessions(username, status, round, score, range_min, range_max, allowed, secret, moves, frost)
  values (me, 'active', p_rounds + 1, p_score, 1, 10, 1, 1, 0, false)
  on conflict (username) do update
    set status = 'active', round = p_rounds + 1, score = p_score, range_min = 1, range_max = 10,
        allowed = 1, secret = 1, moves = 0, frost = false, updated_at = now();
  -- run_finish пишет runs и рекорд и закрывает строку — на этом срабатывает
  -- триггер заданий и достижений
  perform run_finish(me);
  coins := wallet_earn(me, least(20, 2 * p_rounds), 'trial', null);
  if p_rounds > 0 then
    perform quest_event(me, 'play', 1);
    perform play_done(me, 15);
  end if;
  if p_rounds >= 5 then perform weekend_point(me); end if;
  select best_run_score, best_run_rounds into best, best_rounds from students where username = me;
  return jsonb_build_object('best', coalesce(best, 0), 'bestRounds', coalesce(best_rounds, 0), 'coins', coins);
end;
$$;

-- ============ 11. Награды лиг — как своё из коллекции ============
create or replace function set_avatar(p_username text, p_pin text, p_avatar text)
returns text
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; need int;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;
  -- Иконка за обучение — тому, кто его прошёл
  if p_avatar = tutorial_icon() then
    if not coalesce((select tutorial_done from students where username = me), false) then
      raise exception 'avatar_locked';
    end if;
  -- Иконка за серию заданий — тому, у кого была такая серия
  elsif p_avatar = any (quest_icons()) then
    select days into need from quest_rewards() where icon = p_avatar;
    if coalesce((select quest_best from player_stats where username = me), 0) < need then
      raise exception 'avatar_locked';
    end if;
  -- Иконка босса Приключения — тому, кто прошёл его мир
  elsif p_avatar = any (adventure_icons()) then
    need := array_position(adventure_icons(), p_avatar) * 10;
    if coalesce((select adventure from students where username = me), 0) < need then
      raise exception 'avatar_locked';
    end if;
  -- Иконка соперника Приключения — когда весь его мир пройден на 3★
  elsif p_avatar = any (adventure_char_icons()) then
    need := (substr(p_avatar, 5)::int - 1) / 10;
    if substr(coalesce((select adventure_stars from students where username = me), ''), need * 10 + 1, 10) <> '3333333333' then
      raise exception 'avatar_locked';
    end if;
  -- Иконка пути, сундука, магазина или лиги — тому, у кого она есть
  elsif p_avatar = any (pass_icons() || shop_codes('icon') || reward_icons()) then
    if not cosmetic_has(me, 'icon', p_avatar) then raise exception 'avatar_locked'; end if;
  elsif p_avatar is not null and not (p_avatar = any (avatar_choices())) then
    raise exception 'invalid_avatar';
  end if;
  update students set avatar = p_avatar where username = me;
  return p_avatar;
end;
$$;
