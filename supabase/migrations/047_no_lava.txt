-- Убран бонус «Бросок в лаву». Применять в Supabase: SQL Editor → вставить
-- целиком → Run.
--
-- Новые бонусы лавы больше не ставятся: полезные — «два про запас», туман,
-- слепота и спешка, ловушки — как были. В идущих партиях ещё не взятые
-- бонусы лавы заменяются туманом. Если лаву уже взяли до миграции, она
-- доиграет свой один ход — игра это по-прежнему умеет показать.

create or replace function place_match_bonuses(p_match_id bigint)
returns void
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare
  m matches;
  good text[] := array['extra', 'fog', 'blind', 'rush'];
  trap text[] := array['skip', 'gift', 'memory'];
  total int; traps int; i int;
  pool text[] := '{}';
begin
  select * into m from matches where id = p_match_id;
  delete from match_bonuses where match_id = p_match_id and round = m.round;
  if not m.bonuses_on then return; end if;

  total := least(coalesce(m.bonus_count, auto_bonus_count(match_span(m))), match_span(m) - 1);
  total := greatest(1, total);
  traps := round(total / 3.0);

  select array_agg(x order by random()) into good from unnest(good) x;
  select array_agg(x order by random()) into trap from unnest(trap) x;
  for i in 1..(total - traps) loop pool := pool || good[((i - 1) % 4) + 1]; end loop;
  for i in 1..traps loop pool := pool || trap[((i - 1) % 3) + 1]; end loop;
  select array_agg(x order by random()) into pool from unnest(pool) x;

  -- Свободные числа в случайном порядке — первые total из них
  insert into match_bonuses(match_id, round, value, type)
  select p_match_id, m.round, v, pool[row_number() over ()]
  from (
    select v from generate_series(m.range_min, m.range_min + match_span(m) - 1) v
    where v is distinct from m.secret
    order by random()
    limit total
  ) picked;
end;
$$;

-- В идущих партиях: ещё не взятая лава становится туманом
update match_bonuses b set type = 'fog'
from matches m
where b.match_id = m.id and m.status in ('active', 'invited') and b.type = 'lava' and not b.taken;

revoke execute on function place_match_bonuses(bigint) from public, anon;
