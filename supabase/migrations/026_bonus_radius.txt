-- Радиус сигнала «рядом что-то есть».
--
-- Раньше он был ступенькой — 1, 3 или 5 клеток — и не знал, сколько на поле
-- бонусов. На −10…10 два бонуса по три клетки в каждую сторону зажигали сигнал
-- на 57 % прямой: куда ни ходи, «рядом что-то есть», и подсказка ничего не
-- говорила.
--
-- Теперь радиус считается так, чтобы все зоны вместе накрывали не больше трети
-- прямой: k бонусов по 2r клеток — это 2rk, и r = 3n / 20k даёт ровно 30 %.
-- На больших полях бонусов 3 % клеток, и формула сама сходится к пяти — там
-- ничего не меняется. Меняются четыре диапазона: −10…10, −20…20, 1–100 и
-- −100…100.
--
-- Та же формула живёт в браузере (bonusNearRadius) — они сверяются тестом.
-- Целочисленное деление, а не round от дроби: иначе дробь 0,3 в браузере и в
-- базе округлялась бы по-разному ровно на половинках.
create or replace function bonus_near_radius(p_span int)
returns int
language sql
immutable
as $$
  select greatest(1, floor(3 * p_span::numeric / (20 * auto_bonus_count(p_span))))::int;
$$;
