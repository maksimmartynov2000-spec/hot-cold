-- Переписка с друзьями вне игры и вызов друга на рейтинговую партию.
-- Текст здесь такой же готовый, как в матче, и по той же причине: играют дети
-- из разных стран, которых никто не сводил лично, и читают они на разных
-- языках. Код фразы переводится у получателя — швейцарец пишет, а русский
-- читает по-русски. Свободный текст этого не умеет и требует присмотра.

create table if not exists friend_chat (
  id bigint generated always as identity primary key,
  a text not null references students(username) on update cascade on delete cascade,
  b text not null references students(username) on update cascade on delete cascade,
  sender text not null,
  code text not null,
  created_at timestamptz not null default now(),
  constraint friend_chat_ordered check (a < b)
);
create index if not exists friend_chat_idx on friend_chat (a, b, id desc);
alter table friend_chat enable row level security;

-- До какого сообщения человек дочитал. Своя строка на каждую сторону пары:
-- прочитанное одним не делает прочитанным у другого
create table if not exists friend_chat_read (
  username text not null references students(username) on update cascade on delete cascade,
  other text not null references students(username) on update cascade on delete cascade,
  last_id bigint not null default 0,
  primary key (username, other)
);
alter table friend_chat_read enable row level security;

create or replace function friend_phrases()
returns text[]
language sql
immutable
as $$
  select array[
    'hi', 'howru', 'play', 'playranked',
    'later', 'hour', 'tomorrow', 'cantnow',
    'ready', 'yes', 'no', 'thanks',
    'gg', 'strong', 'revenge', 'bye'];
$$;

create or replace function send_friend_phrase(p_username text, p_pin text,
                                              p_to text, p_code text)
returns bigint
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; other text; lo text; hi text; new_id bigint; recent int;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;
  if not (p_code = any(friend_phrases())) then raise exception 'bad_phrase'; end if;

  select s.username into other from students s where s.username_lower = lower(trim(p_to));
  if other is null then raise exception 'no_such_student'; end if;
  if friend_status(me, other) is distinct from 'friend' then raise exception 'not_a_friend'; end if;

  lo := least(me, other); hi := greatest(me, other);

  -- Закидать фразами нельзя: десять за минуту и хватит
  select count(*) into recent from friend_chat
  where a = lo and b = hi and sender = me and created_at > now() - interval '1 minute';
  if recent >= 10 then raise exception 'too_fast'; end if;

  insert into friend_chat(a, b, sender, code) values (lo, hi, me, p_code)
  returning id into new_id;

  -- Держим только полсотни последних: переписка тут нужна, чтобы договориться
  -- об игре, а не чтобы хранить её годами
  delete from friend_chat
  where a = lo and b = hi
    and id <= coalesce((select id from friend_chat c
                        where c.a = lo and c.b = hi
                        order by c.id desc offset 50 limit 1), 0);

  return new_id;
end;
$$;

create or replace function friend_thread(p_username text, p_pin text, p_to text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; mate text; lo text; hi text; rows jsonb; top bigint;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  select s.username into mate from students s where s.username_lower = lower(trim(p_to));
  if mate is null then raise exception 'no_such_student'; end if;
  if friend_status(me, mate) is distinct from 'friend' then raise exception 'not_a_friend'; end if;

  lo := least(me, mate); hi := greatest(me, mate);

  select coalesce(jsonb_agg(jsonb_build_object(
           'id', c.id, 'mine', c.sender = me, 'code', c.code,
           'ago', floor(extract(epoch from (now() - c.created_at)))::int) order by c.id), '[]'::jsonb),
         max(c.id)
    into rows, top
  from friend_chat c where c.a = lo and c.b = hi;

  -- Открыл переписку — значит прочитал
  insert into friend_chat_read(username, other, last_id) values (me, mate, coalesce(top, 0))
  on conflict (username, other) do update set last_id = greatest(friend_chat_read.last_id, excluded.last_id);

  return jsonb_build_object('other', mate, 'messages', rows);
end;
$$;

-- Список друзей теперь говорит, сколько непрочитанного: иначе переписку
-- никто не заметит, пока в неё не зайдёт
drop function if exists list_friends(text, text);
create or replace function list_friends(p_username text, p_pin text)
returns table(username text, relation text, best_run_score int, since timestamptz,
              unread int)
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  return query
  select s.username,
         case
           when f.status = 'accepted' then 'friend'
           when f.requester = me then 'outgoing'
           else 'incoming'
         end,
         s.best_run_score,
         coalesce(f.decided_at, f.created_at),
         (select count(*)::int from friend_chat c
          where c.a = least(me, s.username) and c.b = greatest(me, s.username)
            and c.sender <> me
            and c.id > coalesce((select r.last_id from friend_chat_read r
                                 where r.username = me and r.other = s.username), 0))
  from friendships f
  join students s
    on s.username = case when f.requester = me then f.addressee else f.requester end
  where (f.requester = me or f.addressee = me)
    and f.status in ('accepted', 'pending')
  order by 2, 1;
end;
$$;

-- Вызов друга на рейтинговую партию: условия берутся у выбранной
-- разновидности, а не настраиваются, иначе это уже не рейтинг
create or replace function challenge_friend_ranked(p_username text, p_pin text,
                                                   p_to text, p_mode int)
returns bigint
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; other text; live int; new_id bigint; r record;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;
  if p_mode is null or p_mode < 0 or p_mode > 3 then raise exception 'bad_mode'; end if;

  select s.username into other from students s where s.username_lower = lower(trim(p_to));
  if other is null then raise exception 'no_such_student'; end if;
  if friend_status(me, other) is distinct from 'friend' then raise exception 'not_a_friend'; end if;

  perform expire_stale_matches();

  select count(*) into live from matches
  where status in ('invited', 'active')
    and ((p0 = me and p1 = other) or (p0 = other and p1 = me));
  if live > 0 then raise exception 'match_already_live'; end if;

  select * into r from ranked_rules(p_mode);
  insert into matches(p0, p1, wins_needed, range_min, range_max, frost,
                      bonuses_on, ranked, ranked_mode, status)
  values (me, other, 3, r.range_min, r.range_max, r.frost, r.bonuses,
          true, p_mode, 'invited')
  returning id into new_id;
  return new_id;
end;
$$;


-- Удалённый уносит переписку с собой
create or replace function delete_account(p_username text, p_pin text)
returns text
language plpgsql
security definer
set search_path = public, extensions, pg_temp
as $$
declare me text; gone text; m matches;
begin
  me := check_student_pin(p_username, p_pin);
  if me is null then raise exception 'auth_failed'; end if;

  -- Незаконченные игры нельзя бросить молча: для соперника это победа
  for m in select * from matches
           where status in ('invited', 'active') and (p0 = me or p1 = me) loop
    if m.status = 'active' and m.ranked then
      update matches
      set status = 'finished', match_over = true, round_over = true,
          round_winner = 1 - match_seat(m, me), forfeit_by = match_seat(m, me),
          turn_deadline = null, updated_at = now()
      where id = m.id;
      perform apply_elo(m.id, 1 - match_seat(m, me));
    else
      update matches set status = 'finished', match_over = true, updated_at = now()
      where id = m.id;
    end if;
  end loop;

  delete from friendships where requester = me or addressee = me;
  delete from elo_ratings where username = me;
  delete from ranked_queue where username = me;
  delete from run_sessions where username = me;
  delete from rivalry where a = me or b = me;
  -- Личную переписку удалённый уносит с собой: имя-то переименуется, а вот
  -- разговоры оставлять незачем. Стираем до переименования, иначе пара
  -- перестанет быть упорядоченной и строка не переживёт правку имени
  delete from friend_chat where a = me or b = me;
  delete from friend_chat_read where username = me or other = me;
  -- Одиночные забеги — только его собственная история, она уходит целиком
  delete from runs where username = me;

  gone := '#' || nextval('deleted_player_seq');
  update students
  set username = gone,
      pin_hash = crypt(gen_random_uuid()::text, gen_salt('bf')),
      pin_hint = null,
      best_run_score = 0,
      best_run_rounds = 0,
      failed_logins = 0,
      locked_until = null
  where username = me;

  return gone;
end;
$$;
grant execute on function delete_account(text, text) to anon;

revoke execute on function friend_phrases() from public, anon;
grant execute on function send_friend_phrase(text, text, text, text) to anon;
grant execute on function friend_thread(text, text, text) to anon;
grant execute on function list_friends(text, text) to anon;
grant execute on function challenge_friend_ranked(text, text, text, int) to anon;
