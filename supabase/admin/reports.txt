-- 1. Неразобранные жалобы, новые сверху. В колонке «переписка» — последние
--    сообщения между ними, которые база приложила к жалобе сама
select r.id as "№",
       to_char(r.created_at at time zone 'Asia/Makassar', 'DD.MM HH24:MI') as "когда (Бали)",
       r.reporter as "кто",
       r.reported as "на кого",
       case r.reason when 'rude' then 'грубость' when 'spam' then 'спам' else 'другое' end as "причина",
       coalesce(r.note, '') as "комментарий",
       (select string_agg(m->>'from' || ': ' || coalesce(m->>'text', '[' || (m->>'code') || ']'), E'\n'
                          order by ord)
        from jsonb_array_elements(r.messages) with ordinality t(m, ord)) as "переписка",
       (select count(*) from player_reports x where x.reported = r.reported) as "всего жалоб на него"
from player_reports r
where not r.reviewed
order by r.created_at desc;
